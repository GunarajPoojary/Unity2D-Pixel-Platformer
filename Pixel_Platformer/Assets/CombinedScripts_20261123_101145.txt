using UnityEngine;
using UnityEngine.EventSystems;

namespace PixelPlatformer
{
    public class Bootstrapper : MonoBehaviour
    {
        [SerializeField] private SceneReference _mainMenuScene;

        [Header("Prefabs")]
        [SerializeField] private EventSystem _eventSystemPrefab;

        [SerializeField] private AudioManager _audioManagerPrefab;
        [SerializeField] private GameManager _gameManagerPrefab;
        [SerializeField] private ProgressionManager _gameProgressDataManagerPrefab;
        [SerializeField] private SceneController _sceneControllerPrefab;

        private AudioManager _audioManager;
        private SceneController _sceneController;
        private GameManager _gameManager;
        private ProgressionManager _gameProgressDataManager;

        private async void Start()
        {
            CreateObjects(); // Instantiate core Game Objects

            // Initialize managers
            SetupObjects();

            // load main Menu Music to memory
            await _gameManager.PreloadMainMenuMusic();

            await _sceneController.LoadMainMenu();

            _gameManager.PlayMainMenuMusic();
        }

        private void SetupObjects()
        {
            _audioManager.Initialize();
            _gameManager.Setup(_sceneController);
            _gameProgressDataManager.Init(_gameManager.TotalLevels);
        }

        private void CreateObjects()
        {
            Transform persistentRoot = new GameObject("Persistent Objects").transform;

            if (EventSystem.current != null)
                Destroy(EventSystem.current.gameObject);

            Instantiate(_eventSystemPrefab, persistentRoot);

            _sceneController = Instantiate(_sceneControllerPrefab, persistentRoot);
            _audioManager = Instantiate(_audioManagerPrefab, persistentRoot);
            _gameManager = Instantiate(_gameManagerPrefab, persistentRoot);
            _gameProgressDataManager = Instantiate(_gameProgressDataManagerPrefab, persistentRoot);

            DontDestroyOnLoad(persistentRoot);
        }
    }
}

using System;
using System.Collections.Generic;

namespace PixelPlatformer
{
    public readonly struct FruitCollectedEvent { }
    public readonly struct CameraShakeEventData { }
    public readonly struct StepOnStartCheckpointEvenData { }
    public readonly struct PlayerDiedEventData { }
    public readonly struct StompedEnemyEventData { }
    public readonly struct LevelCompletedEvent
    {
        public readonly LevelResult levelResult;

        public LevelCompletedEvent(LevelResult result)
        {
            levelResult = result;
        }
    }

    public static class GameEvents
    {
        private static readonly Dictionary<Type, List<Delegate>> _subscribers = new();

        public static void Subscribe<T>(Action<T> listener)
        {
            Type eventType = typeof(T); // key

            if (_subscribers.TryGetValue(eventType, out var listeners))
            {
                listeners.Add(listener);
            }
            else
            {
                _subscribers[eventType] = new List<Delegate> { listener };
            }
        }

        public static void Unsubscribe<T>(Action<T> listener)
        {
            Type eventType = typeof(T); // key

            if (_subscribers.TryGetValue(eventType, out var listeners))
            {
                listeners.Remove(listener); // remove listener if it has

                // there are no listeners then remove the key aswell
                if (listeners.Count == 0)
                {
                    _subscribers.Remove(eventType);
                }
            }
        }

        public static void Publish<T>(T eventData)
        {
            Type eventType = typeof(T); // key

            if (_subscribers.TryGetValue(eventType, out List<Delegate> listeners))
            {
                for (int i = 0; i < listeners.Count; i++)
                {
                    ((Action<T>)listeners[i])?.Invoke(eventData);
                }
            }
        }

        public static void Clear()
        {
            _subscribers.Clear();
        }
    }
}

using Cysharp.Threading.Tasks;
using UnityEditor;
using UnityEngine;

namespace PixelPlatformer
{
    public class GameManager : MonoBehaviour
    {
        public static GameManager Instance { get; private set; }

        [SerializeField] private LevelDefinitionSO[] _levels;

        [Header("Music")]
        [SerializeField] private AudioClip _mainMenuMusic;

        private int _currentLevelIndex;
        private SceneController _sceneController;

        private ProgressionManager _progressionManager;
        private GameplayManager _gameplayManager;
        private PlayerManager _playerManager;
        private AudioManager _audioManager;

        public int TotalLevels { get { return _levels.Length; } }

        private void OnEnable()
        {
            GameEvents.Subscribe<LevelCompletedEvent>(HandleLevelCompleted);
        }

        private void OnDisable()
        {
            GameEvents.Unsubscribe<LevelCompletedEvent>(HandleLevelCompleted);
        }

        private void OnDestroy()
        {
            GameEvents.Clear();
        }

        public void Setup(SceneController sceneController)
        {
            Instance = this;

            _sceneController = sceneController;
            _progressionManager = ProgressionManager.Instance;
            _audioManager = AudioManager.Instance;
        }

        public void QuitGame()
        {
#if UNITY_EDITOR
            EditorApplication.isPlaying = false;
#else
            Application.Quit();
#endif
        }



        public async void GoToMainMenu()
        {
            if (_gameplayManager != null)
                _gameplayManager.EndSession();

            await _sceneController.GoToMainMenu();

            _gameplayManager = null;
            _playerManager = null;

            PlayMainMenuMusic();
        }

        public async void GoToNextLevel()
        {
            int nextLevelIndex = _currentLevelIndex + 1;

            // the current level was the final level
            if (nextLevelIndex > TotalLevels)
            {
                await _sceneController.GoToMainMenu();
                PlayMainMenuMusic();
                return;
            }

            await _sceneController.GoToNextLevel(
                nextLevelIndex,
                onUnloadScene: () =>
                {
                    _gameplayManager.EndSession();
                    _playerManager.DespawnPlayer();
                },
                onLoadScene: () =>
                {
                    _currentLevelIndex = nextLevelIndex;
                    _gameplayManager.Initialize(_currentLevelIndex);
                });

            _gameplayManager.StartSession();
        }

        public async void StartLevel(int levelIndex)
        {
            if (levelIndex < 1 || levelIndex > TotalLevels)
            {
                Debug.LogError($"Invalid level index: {levelIndex}");
                return;
            }

            _progressionManager = ProgressionManager.Instance;
            _audioManager = AudioManager.Instance;

            _audioManager.StopMusic();

            await _sceneController.LoadLevel(
                levelIndex,
                onLoadGameplay: () =>
                {
                    _gameplayManager = GameplayManager.Instance;
                    _playerManager = PlayerManager.Instance;
                },
                onLoadLevel: () =>
                {
                    _currentLevelIndex = levelIndex;
                    _gameplayManager.Initialize(_currentLevelIndex);
                });

            _gameplayManager.StartSession();
        }

        public async void RestartLevel()
        {
            if (_gameplayManager == null)
                _gameplayManager = GameplayManager.Instance;

            await _sceneController.ReloadActiveScene(
                onUnloadScene: () =>
                {
                    _gameplayManager.EndSession();

                    UIManager.Instance.HideLoseScreen();
                    UIManager.Instance.HideWinScreen();

                    _playerManager.DespawnPlayer();
                },
                onLoadScene: () =>
                {
                    _gameplayManager.Initialize(_currentLevelIndex);
                });

            _gameplayManager.StartSession();
        }


        public Sprite GetLevelIcon(int levelIndex)
        {
            int index = levelIndex - 1;

            if (index < 0 || index >= TotalLevels)
            {
                Debug.LogError($"Cannot get icon. Invalid level index: {levelIndex}");
                return null;
            }

            return _levels[index].LevelIcon;
        }


        public async UniTask PreloadMainMenuMusic()
        {
            if (_mainMenuMusic == null)
                return;

            await UniTask.WaitUntil(() =>
                _mainMenuMusic.LoadAudioData() == true);
        }

        public void PlayMainMenuMusic()
        {
            _audioManager = AudioManager.Instance;

            if (_audioManager != null && _mainMenuMusic != null)
                _audioManager.PlayMusic(_mainMenuMusic);
        }

        private void HandleLevelCompleted(LevelCompletedEvent e)
        {
            if (_progressionManager == null)
                _progressionManager = ProgressionManager.Instance;

            if (_currentLevelIndex < 1 || _currentLevelIndex > TotalLevels)
                return;

            int levelIndex = _levels[_currentLevelIndex - 1].LevelIndex;

            _progressionManager.SetLevelRecordData(levelIndex, e.levelResult);
        }
    }
}

using System;
using System.Collections.Generic;

namespace PixelPlatformer
{
    [Serializable]
    public class ProgressData
    {
        public List<LevelRecord> records;

        public ProgressData(int totalLevels)
        {
            records = new List<LevelRecord>(totalLevels);



            for (int i = 0; i < totalLevels; i++)
            {
                records.Add(new LevelRecord
                {
                    levelIndex = i + 1,
                    isUnlocked = i == 0,   // level 1 unlocked by default
                    StarsEarned = 0,
                    bestScore = 0f
                });
            }
        }
    }
}

using System;
using System.Collections.Generic;
using UnityEngine;

namespace PixelPlatformer
{
    public class ProgressionManager : Singleton<ProgressionManager>
    {
        private const string GAME_PROGRESS_KEY = "Game_Progress";

        private ProgressData _progressData;
        private int _totalLevels;

        public event Action OnDataErased;

        private void OnApplicationPause(bool paused)
        {
            if (paused) SaveData();
        }

        private void OnApplicationQuit()
        {
            SaveData();
        }

        private void OnDestroy()
        {
            SaveData();
        }

        public void Init(int totalLevels)
        {
            _totalLevels = totalLevels;

            string json = PlayerPrefs.GetString(GAME_PROGRESS_KEY, string.Empty);

            if (!string.IsNullOrEmpty(json))
                _progressData = JsonUtility.FromJson<ProgressData>(json);

            if (_progressData == null || _progressData.records == null || _progressData.records.Count == 0)
                _progressData = new ProgressData(totalLevels);
        }

        public List<LevelRecord> LoadData()
        {
            return _progressData.records;
        }

        public void SaveData()
        {
            PlayerPrefs.SetString(GAME_PROGRESS_KEY, JsonUtility.ToJson(_progressData));
            PlayerPrefs.Save();
        }

        public void SetLevelRecordData(int levelIndex, LevelResult result)
        {
            var index = levelIndex - 1;
            LevelRecord record = _progressData.records[index];

            // if recorded data found then override it by new best record
            record.isUnlocked = true;
            record.StarsEarned = Mathf.Max(record.StarsEarned, result.starsEarned);
            record.bestScore = Mathf.Max(record.bestScore, result.score);

            // also unlock the next level with new default record data
            var nextLevelIndex = levelIndex + 1;
            UnlockLevel(nextLevelIndex);
            SaveData();
        }

        public void UnlockLevel(int targetLevelIndex)
        {
            if (targetLevelIndex > _progressData.records.Count) return;

            var index = targetLevelIndex - 1;
            var record = _progressData.records[index];
            if (record.isUnlocked) return;

            record.isUnlocked = true;
            record.StarsEarned = 0;
            record.bestScore = 0;
        }

        public void EraseData()
        {
            PlayerPrefs.DeleteKey(GAME_PROGRESS_KEY);
            _progressData = new ProgressData(_totalLevels);
            SaveData();
            OnDataErased?.Invoke();
        }

        public LevelRecord GetRecord(int currentLevelIndex)
        {
            foreach (var item in _progressData.records)
            {
                if (item.levelIndex == currentLevelIndex)
                    return item;
            }

            return null;
        }
    }
}

using System;
using Cysharp.Threading.Tasks;
using UnityEngine;
using UnityEngine.SceneManagement;

namespace PixelPlatformer
{
    public class SceneController : MonoBehaviour
    {
        [SerializeField] private LoadingScreen _loadingScreen;
        [SerializeField] private SceneTransitionFX _sceneTransitionFX;

        [SerializeField] private SceneReference _mainMenuScene;
        [SerializeField] private SceneReference _gameplayScene;

        [SerializeField] private LevelDefinitionSO[] _levels;

        private void Awake()
        {
            _sceneTransitionFX.ResetState();
        }

        #region Public API
        public async UniTask LoadLevel(int levelIndex, Action onLoadGameplay, Action onLoadLevel)
        {
            await EnterLoadingScreen();

            _loadingScreen.SetProgress(0.1f);

            // load gameplay
            AsyncOperation gameplayLoadOperation = await LoadScene(_gameplayScene.BuildIndex);

            _loadingScreen.SetProgress(0.2f);

            await ActivateScene(gameplayLoadOperation);

            _loadingScreen.SetProgress(0.3f);

            // unload main menu
            await UnloadScene(_mainMenuScene.BuildIndex);

            _loadingScreen.SetProgress(0.4f);



            // get the gameplay and player manager from Gameplay Scene
            onLoadGameplay?.Invoke();


            // load saved level
            var index = levelIndex - 1;

            AsyncOperation levelLoadOperation = await LoadScene(_levels[index].LevelScene.BuildIndex);

            _loadingScreen.SetProgress(0.5f);

            await ActivateScene(levelLoadOperation);

            _loadingScreen.SetProgress(0.6f);


            var sceneBuildIndex = _levels[index].LevelScene.BuildIndex;

            SetActiveScene(sceneBuildIndex);




            onLoadLevel?.Invoke();

            _loadingScreen.SetProgress(1f);


            await ExitLoadingScreen();
        }

        public async UniTask LoadMainMenu()
        {
            ShowLoadingScreen();
            _loadingScreen.SetProgress(0.2f);

            AsyncOperation menuLoadOperation = await LoadScene(_mainMenuScene.BuildIndex);

            _loadingScreen.SetProgress(0.4f);

            await ActivateScene(menuLoadOperation);

            _loadingScreen.SetProgress(0.8f);

            await UnloadScene(0);

            _loadingScreen.SetProgress(1f);

            await ExitLoadingScreen();
        }

        public async UniTask<AsyncOperation> LoadScene(int sceneIndex)
        {
            AsyncOperation op = SceneManager.LoadSceneAsync(sceneIndex, LoadSceneMode.Additive);
            op.allowSceneActivation = false;

            while (op.progress < 0.9f)
                await UniTask.Yield(); // jumps to next frame

            return op;
        }

        public async UniTask GoToMainMenu()
        {
            await EnterLoadingScreen();

            _loadingScreen.SetProgress(0.1f);

            // unload active level scene
            await UnloadActiveScene();

            _loadingScreen.SetProgress(0.3f);

            // load main menu
            AsyncOperation op = await LoadScene(_mainMenuScene.BuildIndex);

            _loadingScreen.SetProgress(0.5f);
            await ActivateScene(op);

            // unload gameply scene
            await UnloadScene(_gameplayScene.BuildIndex);

            _loadingScreen.SetProgress(0.8f);

            SetActiveScene(_mainMenuScene.BuildIndex);

            _loadingScreen.SetProgress(1f);

            await ExitLoadingScreen();
        }

        public async UniTask GoToNextLevel(int levelIndex, Action onUnloadScene, Action onLoadScene)
        {
            await EnterLoadingScreen();

            await UnloadActiveScene();
            onUnloadScene?.Invoke();

            _loadingScreen.SetProgress(0.3f);


            bool hasNextLevel = levelIndex < _levels.Length;

            if (!hasNextLevel)
            {
                // there are no more levels, send player back to the main menu
                HideLoadingScreen();
                return;
            }

            var index = levelIndex - 1;

            AsyncOperation op = await LoadScene(_levels[index].LevelScene.BuildIndex);
            await ActivateScene(op);

            _loadingScreen.SetProgress(1f);

            SetActiveScene(_levels[index].LevelScene.BuildIndex);
            onLoadScene?.Invoke();

            await ExitLoadingScreen();
        }














        public void ShowLoadingScreen()
        {
            _loadingScreen.Show();
        }

        public void HideLoadingScreen()
        {
            _loadingScreen.Hide();
        }

        public async UniTask UnloadScene(int sceneIndex)
        {
            AsyncOperation op = SceneManager.UnloadSceneAsync(sceneIndex);

            if (op == null)
            {
                Debug.LogError($"Could not unload scene {sceneIndex} — not loaded or already unloading.");
                return;
            }

            await op.ToUniTask();
        }

        public async UniTask ActivateScene(AsyncOperation operation)
        {
            operation.allowSceneActivation = true;

            await operation.ToUniTask(); // convert AsyncOperation to UniTask which can be await
        }


        public async UniTask EnterLoadingScreen(Action onTransitionInComplete = null)
        {
            await _sceneTransitionFX.PlayIn(); // slowly cover the screen by transition fx
            _loadingScreen.Show();  // show loading screen
            onTransitionInComplete?.Invoke(); // Hide other UI elements
            await _sceneTransitionFX.PlayOut(); // slowly uncover the screen using fx
        }

        public async UniTask ExitLoadingScreen(Action onTransitionOutComplete = null)
        {
            await _sceneTransitionFX.PlayIn(); // slowly cover the screen using fx
            onTransitionOutComplete?.Invoke(); // show other UI element or activate scene
            _loadingScreen.Hide(); // Hide loading screen
            await _sceneTransitionFX.PlayOut(); // slowly uncover the screen using fx
        }

        public void SetActiveScene(int buildIndex)
        {
            Scene scene = SceneManager.GetSceneByBuildIndex(buildIndex);

            SceneManager.SetActiveScene(scene);
        }

        public async UniTask ReloadActiveScene(Action onUnloadScene, Action onLoadScene)
        {
            await EnterLoadingScreen();

            _loadingScreen.SetProgress(0.3f);








            Scene activeScene = SceneManager.GetActiveScene();
            int level = activeScene.buildIndex;

            if (!activeScene.IsValid() || !activeScene.isLoaded)
                throw new Exception("Scene is invalid");

            await UnloadScene(activeScene.buildIndex);
            onUnloadScene?.Invoke();
            _loadingScreen.SetProgress(0.4f);

            // load the same level again
            AsyncOperation op = await LoadScene(level);
            _loadingScreen.SetProgress(0.6f);

            await ActivateScene(op);

            SetActiveScene(level);






            onLoadScene?.Invoke();

            _loadingScreen.SetProgress(1f);

            await ExitLoadingScreen();
        }


        public async UniTask UnloadActiveScene()
        {
            Scene levelScene = SceneManager.GetActiveScene();

            if (levelScene.IsValid() && levelScene.isLoaded)
                await UnloadScene(levelScene.buildIndex);
        }
        #endregion
    }
}

using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Rigidbody2D))]
    public class FatBirdController : MonoBehaviour
    {
        private static readonly int FallID = Animator.StringToHash("isFalling");
        private static readonly int HitID = Animator.StringToHash("hit");

        [Header("Patrol")]
        [SerializeField] private float _flySpeed = 0.5f;
        [SerializeField] private float _patrolOffsetY;
        [SerializeField] private float _patrolHeight = 4f;
        [SerializeField] private float _bodyHalfHeight = 0.5f;

        [Header("Detection")]
        [SerializeField] private LayerMask _playerMask;
        [SerializeField] private LayerMask _groundLayer;
        [SerializeField] private float _detectWidth = 1f;
        [SerializeField] private float _detectRange = 8f;

        [Header("Drop")]
        [SerializeField] private float _fallGravityScale = 4f;

        private Rigidbody2D _rb;
        private Animator _animator;
        private EnemyHitter _hitHandler;

        private float _patrolCenterY;
        private float _patrolDir = 1f;    // 1f means going up, -1f means going down
        private bool _dropping;      

        private void Awake()
        {
            _rb = GetComponent<Rigidbody2D>();
            _animator = GetComponentInChildren<Animator>();
            _hitHandler = new EnemyHitter(_playerMask);
        }

        private void Start()
        {
            _patrolCenterY = transform.position.y + _patrolOffsetY;
            Move(1f);
        }

        private void FixedUpdate()
        {
            if (_dropping)
                return;

            if (IsPlayerBelow())
            {
                _dropping = true;
                SetFalling(true);
            }
            else
            {
                Patrol();
            }
        }

        private void Patrol()
        {
            float limit = _patrolCenterY + _patrolDir * (_patrolHeight * 0.5f - _bodyHalfHeight);

            if (_patrolDir * (_rb.position.y - limit) < 0f)
                return;

            _rb.position = new Vector2(_rb.position.x, limit);
            Move(-_patrolDir);
        }

        private void Move(float dir)
        {
            _patrolDir = dir;
            _rb.linearVelocity = new Vector2(0f, dir * _flySpeed);
        }

        private bool IsPlayerBelow()
        {
            Vector2 origin = new Vector2(_rb.position.x, _rb.position.y - _bodyHalfHeight);
            RaycastHit2D hit = Physics2D.BoxCast(origin, new Vector2(_detectWidth, 0.05f), 0f,
                Vector2.down, _detectRange, _playerMask | _groundLayer);

            return hit.collider != null && InMask(_playerMask, hit.collider);
        }

        private void SetFalling(bool falling)
        {
            _rb.linearVelocity = Vector2.zero;
            _rb.gravityScale = falling ? _fallGravityScale : 0f;
            _animator.SetBool(FallID, falling);
        }

        public void Rise()
        {
            _dropping = false;
            Move(1f);
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            _hitHandler.TryHitTarget(col, transform.position);

            if (InMask(_playerMask, col.collider))
                _animator.SetTrigger(HitID);

            if (_dropping && InMask(_groundLayer, col.collider) && col.GetContact(0).normal.y > 0.5f)
                SetFalling(false);
        }

        private static bool InMask(LayerMask mask, Collider2D c) =>
            (mask.value & (1 << c.gameObject.layer)) != 0;

        private void OnDrawGizmosSelected()
        {
            float centerY = Application.isPlaying ? _patrolCenterY : transform.position.y + _patrolOffsetY;
            float x = transform.position.x;
            Vector3 bottom = new Vector3(x, centerY - _patrolHeight * 0.5f, 0f);
            Vector3 top = new Vector3(x, centerY + _patrolHeight * 0.5f, 0f);

            Gizmos.color = Color.green;
            Gizmos.DrawLine(bottom, top);
            Gizmos.DrawWireSphere(bottom, 0.15f);
            Gizmos.DrawWireSphere(top, 0.15f);

            Gizmos.color = Color.red;
            Gizmos.DrawWireCube(new Vector3(x, transform.position.y - _bodyHalfHeight - _detectRange * 0.5f, 0f),
                new Vector3(_detectWidth, _detectRange, 0f));
        }
    }
}

    [RequireComponent(typeof(SpriteRenderer), typeof(Hittable), typeof(Collider2D))]
    public class GhostController : MonoBehaviour

        [SerializeField] private float _runSpeed = 3f;
        [SerializeField] private LayerMask _playerMask;
        [SerializeField] private float _idleTime = 1f;

        [SerializeField] private float _visibleTime = 3f;

            _collider = GetComponent<Collider2D>();
            _hitHandler = new EnemyHitter(_playerMask);

            _idleClip._loop = true;
            SetupClip(_idleClip);
            SetupClip(_appearClip);
            SetupClip(_disappearClip);
            SetupClip(_hitClip);

            _stateTimer = _visibleTime;
        }

        private void Start()
        {
            _patrolCenterX = transform.position.x + _patrolOffsetX;
            _idleClip.PlayClip();
        }

        private void SetupClip(SingleClipAnimator clip)
        {
            clip.monoBehaviour = this;

            if (clip._renderer == null)
                clip._renderer = _renderer;
        }

        private void StopAllClips()
        {
            _idleClip.StopClip();
            _appearClip.StopClip();
            _disappearClip.StopClip();
            _hitClip.StopClip();
        }

        private void OnEnable()
        {
            _hittable.OnHit += HandleHit;
        }

        private void OnDisable()
        {
            _hittable.OnHit -= HandleHit;
        }

        private void HandleHit()
        {
            StopAllClips();
            _hitClip.PlayClip();
            enabled = false;

        private void Update()
        {
            UpdateVisibility();

            if (_shouldIdle)
            {
                HandleIdling();
                return;
            }

            Run();
            CheckPatrolEnd();
        }

        private void UpdateVisibility()
        {
            if (_isTransitioning) return;

            _stateTimer -= Time.deltaTime;

            if (_stateTimer > 0f) return;

            if (_isHidden)
                Appear();
            else
                Disappear();
        }

        private void Disappear()
        {
            _isTransitioning = true;
            _collider.enabled = false;

            StopAllClips();
            _disappearClip.PlayClip(OnDisappeared);
        }

        private void OnDisappeared()
        {
            _isTransitioning = false;
            _isHidden = true;
            _stateTimer = _hiddenTime;
            _renderer.sprite = null;
        }

        private void Appear()
        {
            _isTransitioning = true;
            _collider.enabled = true;

            StopAllClips();
            _appearClip.PlayClip(OnAppeared);
        }

        private void OnAppeared()
        {
            _isTransitioning = false;
            _isHidden = false;
            _stateTimer = _visibleTime;
            _idleClip.PlayClip();
        }

        private void Run()
        {
            Vector3 direction = IsFacingLeft ? Vector3.left : Vector3.right;
            transform.position += _runSpeed * Time.deltaTime * direction;
        }

        private void CheckPatrolEnd()
        {
            float minX = _patrolCenterX - _patrolWidth * 0.5f + _bodyHalfWidth;
            float maxX = _patrolCenterX + _patrolWidth * 0.5f - _bodyHalfWidth;

            Vector3 position = transform.position;

            if (IsFacingLeft && position.x <= minX)
            {
                position.x = minX;
                transform.position = position;
                StartIdling();
            }
            else if (!IsFacingLeft && position.x >= maxX)
            {
                position.x = maxX;
                transform.position = position;
                StartIdling();
            }
        }

        private void StartIdling()
        {
            _shouldIdle = true;
            _idleTimer = _idleTime;
        }

        private void HandleIdling()
        {
            _idleTimer -= Time.deltaTime;

            if (_idleTimer > 0f) return;

            _renderer.flipX = !_renderer.flipX;
            _shouldIdle = false;
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            _hitHandler.TryHitTarget(col, transform.position);
        }

        private void OnDrawGizmosSelected()
        {
            float centerX = Application.isPlaying ? _patrolCenterX : transform.position.x + _patrolOffsetX;
            float y = transform.position.y;

            Vector3 left = new Vector3(centerX - _patrolWidth * 0.5f, y, 0f);
            Vector3 right = new Vector3(centerX + _patrolWidth * 0.5f, y, 0f);

            Gizmos.color = Color.green;
            Gizmos.DrawLine(left, right);
            Gizmos.DrawWireSphere(left, 0.15f);
            Gizmos.DrawWireSphere(right, 0.15f);
        }

using System;
using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Rigidbody2D), typeof(SpriteRenderer))]
    public class BulletChunk : MonoBehaviour
    {
        [SerializeField] private LayerMask _groundMask;
        [SerializeField] private float _blinkDuration = 1f;
        [SerializeField] private float _blinkInterval = 0.08f;

        [SerializeField] private Rigidbody2D _rb;
        [SerializeField] private SpriteRenderer _renderer;

        private Sprite _sprite;
        private bool _hasLanded;

        public event Action<BulletChunk> Finished;

        private void OnCollisionEnter2D(Collision2D col)
        {
            if (!_hasLanded && (_groundMask.value & (1 << col.gameObject.layer)) != 0)
                _hasLanded = true;
        }

        public void Initialize()
        {
            _sprite = _renderer.sprite;
        }

        public void Launch(Vector2 velocity, float spin)
        {
            _hasLanded = false;

            gameObject.SetActive(true);

            _renderer.sprite = _sprite;
            _rb.linearVelocity = velocity;
            _rb.angularVelocity = spin;

            StartCoroutine(BlinkRoutine());
        }

        public void ResetState()
        {
            StopAllCoroutines();
            _hasLanded = false;

            if (_sprite != null)
                _renderer.sprite = _sprite;

            _rb.linearVelocity = Vector2.zero;
            _rb.angularVelocity = 0f;

            transform.SetLocalPositionAndRotation(Vector3.zero, Quaternion.identity);

            gameObject.SetActive(false);
        }

        private IEnumerator BlinkRoutine()
        {
            while (!_hasLanded)
                yield return null;

            float totalTime = 0f;
            float elapsedTime = 0f;
            bool visible = true;

            while (totalTime < _blinkDuration)
            {
                float deltaTime = Time.deltaTime;
                totalTime += deltaTime;
                elapsedTime += deltaTime;

                if (elapsedTime >= _blinkInterval)
                {
                    visible = !visible;

                    if (visible)
                        _renderer.sprite = _sprite;
                    else
                        _renderer.sprite = null;

                    elapsedTime -= _blinkInterval;
                }

                yield return null;
            }

            ResetState();

            Finished?.Invoke(this);
        }
    }
}

            _renderer = GetComponent<SpriteRenderer>();

            _hitHandler.TryHitTarget(col, transform.position);

            _animator.SetBool(AttackID, false);
            _animator.SetTrigger(HitID);
            enabled = false;
        }

        {
            Vector2 origin = (Vector2)transform.position + MuzzleOffset();
            RaycastHit2D hit = Physics2D.BoxCast(origin, new Vector2(0.05f, _detectHeight), 0f,
                FacingDir, _detectRange, _playerMask);

            return hit.collider != null && (_playerMask.value & (1 << hit.collider.gameObject.layer)) != 0;

        {
            Vector3 pos = (Vector2)transform.position + MuzzleOffset();

            Bullet bullet = _bulletPool.GetBullet(pos, Quaternion.identity);

            Gizmos.DrawWireCube(origin + dir * (_detectRange * 0.5f), new Vector3(_detectRange, _detectHeight, 0f));
            Gizmos.DrawWireSphere(origin, 0.1f);
        }

using System;
using DG.Tweening;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Animator), typeof(SpriteRenderer))]
    public class RhinoController : MonoBehaviour, IHittable
    {
        [SerializeField] private float _chaseSpeed = 5f;
        [SerializeField] private float _acceleration = 5f;
        [SerializeField] private LayerMask _enemyMask;
        [SerializeField] private LayerMask _wallMask;
        [SerializeField] private float _height = 1f;
        [SerializeField] private float _fallbackCastDistance = 50f;
        [SerializeField] private float _wallStopDistance = 0.01f;
        [SerializeField] private float _lOSRadius = 0.1f;


        [SerializeField] private float _frontOffset = 0.1f;
        [SerializeField] private float _backOffset = 0.1f;

        private SpriteRenderer _renderer;
        private float _currentVelocity;
        private Vector2 _front;
        private Vector2 _leftWallContactPoint;
        private Vector2 _rightWallContactPoint;
        private bool _isChase;
        private bool _canDetect = true;

        [Header("Jump tween")]
        [SerializeField] private float _jumpDistance = 2.5f;
        [SerializeField] private float _jumpPower = 03f;
        [SerializeField] private float _jumpDuration = 0.5f;
        private Vector2 _back;
        private Vector2 _direction;
        // private Vector2 _chaseDirection;
        private bool _isFacingRight;

        private bool _isKillable = true;

        public bool IsHittable
        {
            get
            {
                return _isKillable;
            }
        }

        public event Action OnRunStart;
        public event Action OnRunStop;
        public event Action OnJumpStart;
        public event Action OnJumpEnd;
        public event Action OnDie;

        private void Awake()
        {
            _renderer = GetComponent<SpriteRenderer>();
        }

        private void Start()
        {
            InitWallContactPoints();
            _isFacingRight = _renderer.flipX;
        }

        // initialize line of sight values
        // idle: patroling using line of sight on both sides
        // chase: player detected then start running untill hits wall
        // recaclulate line of sight values
        // repeat  

        private void InitWallContactPoints()
        {
            _direction = _isFacingRight ? Vector2.right : Vector2.left; // if true then facing screen right(+x)

            _front = (Vector2)transform.position + (_direction * _frontOffset);
            _front.y += _height;

            _back = (Vector2)transform.position + (-_direction * _backOffset);
            _back.y += _height;

            _leftWallContactPoint = GetWallContactPoint(_front, _direction);
            _rightWallContactPoint = GetWallContactPoint(_back, -_direction);
        }

        private Vector2 GetWallContactPoint(Vector2 origin, Vector2 direction)
        {
            RaycastHit2D wallHit = Physics2D.Raycast(origin, direction, _fallbackCastDistance, _wallMask);

            return wallHit.point;
        }

        private void Update()
        {
            if (_canDetect)
                HandleLOS();

            HandleChase();
        }

        private void HandleLOS()
        {
            // var direction = _isFacingRight ? Vector2.right : Vector2.left;
            // var wallContactPoint = Vector2.zero;

            // line cast start point which is face
            _front = (Vector2)transform.position + ((_isFacingRight ? Vector2.right : Vector2.left) * _frontOffset);
            _front.y += _height;

            var isEnemyFront = Physics2D.Linecast(_front, _isFacingRight ? _rightWallContactPoint : _leftWallContactPoint, _enemyMask).collider != null;

            if (isEnemyFront)
            {
                Debug.Log("Start chasing");
                // _chaseDirection = direction;
                _canDetect = false;
                _isChase = true;
                OnRunStart?.Invoke();
                return;
            }

            _back = (Vector2)transform.position + ((_isFacingRight ? Vector2.left : Vector2.right) * _backOffset);
            _back.y += _height;

            var isEnemyBehind = Physics2D.Linecast(_back, _isFacingRight ? _leftWallContactPoint : _rightWallContactPoint, _enemyMask).collider != null;

            if (isEnemyBehind)
            {
                Debug.Log("Start chasing");
                // _chaseDirection = -_direction;
                _canDetect = false;
                Turn();
                _isChase = true;
                OnRunStart?.Invoke();
            }
        }
        private void Turn()
        {
            _renderer.flipX = !_renderer.flipX;
            _isFacingRight = _renderer.flipX;
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            if (col.collider == null) return;

            if ((_wallMask.value & (1 << col.gameObject.layer)) != 0)
            {
                Debug.Log("Contact with wall");
            }
            // if ((_enemyMask.value & (1 << col.gameObject.layer)) != 0
            //     && col.collider.TryGetComponent<IImpactable>(out var crusher))
            // {
            //     crusher.ApplyImpact(_pushForce);
            // }
            if ((_enemyMask.value & (1 << col.gameObject.layer)) != 0
                && col.collider.TryGetComponent<IHittable>(out var damageable))
            {
                if (damageable == null || !damageable.IsHittable) return;

                var contactPoint = col.contacts[0].point;

                Vector2 currentPos = transform.position;

                Vector2 direction = contactPoint - currentPos;

                if (Mathf.Abs(direction.x) > Mathf.Abs(direction.y))
                {
                    damageable.TakeHit(transform.position);
                    _canDetect = false;
                }
            }
        }

        private void HandleChase()
        {
            if (!_isChase) return;

            _currentVelocity = Mathf.MoveTowards(_currentVelocity, _chaseSpeed, _acceleration * Time.deltaTime);
            float moveAmount = _currentVelocity * Time.deltaTime;

            ApplyMovement((_isFacingRight ? Vector2.right : Vector2.left) * moveAmount);

            var frontX = transform.position.x + ((_isFacingRight ? 1 : -1) * _frontOffset);

            if (!_renderer.flipX)
            {
                if (frontX - _leftWallContactPoint.x < _wallStopDistance)
                {
                    Debug.Log("Hit wall");

                    HandleHitWall();
                }
            }
            else
            {
                if (_rightWallContactPoint.x - frontX < _wallStopDistance)
                {
                    Debug.Log("Hit wall");

                    HandleHitWall();
                }
            }
        }

        private void HandleHitWall()
        {
            _isChase = false;
            OnRunStop?.Invoke();

            GameEvents.Publish(new CameraShakeEventData());
            _currentVelocity = 0;
            ApplyMovement(Vector2.zero);

            // var jump = (_renderer.flipX ? Vector2.left : Vector2.right) * _jumpDistance;
            // jump.y = transform.position.y;
            // jump.x = _renderer.flipX ? jump.x - transform.position.x : jump.x + transform.position.x;


            var jump = (Vector2)transform.position;
            jump.x = _renderer.flipX ? jump.x - _jumpDistance : jump.x + _jumpDistance;
            OnJumpStart?.Invoke();

            // var debugObj = GameObject.CreatePrimitive(PrimitiveType.Sphere);
            // debugObj.transform.position = jump;
            transform.DOKill();
            transform.DOJump(jump, _jumpPower, 1, _jumpDuration).OnComplete(() =>
    {
        _canDetect = true;
        OnJumpEnd?.Invoke();
    });
        }

        private void ApplyMovement(Vector2 movement)
        {
            transform.position += (Vector3)movement;
        }

        private void OnDrawGizmos()
        {
            if (!Application.isPlaying) return;

            Gizmos.color = Color.red;

            Gizmos.DrawWireSphere(_front, _lOSRadius);

            Gizmos.DrawWireSphere(_isFacingRight ? _rightWallContactPoint : _leftWallContactPoint, _lOSRadius);
            Gizmos.DrawLine(_front, _isFacingRight ? _rightWallContactPoint : _leftWallContactPoint);

            Gizmos.DrawWireSphere(_back, _lOSRadius);

            Gizmos.DrawWireSphere(_isFacingRight ? _leftWallContactPoint : _rightWallContactPoint, _lOSRadius);
            Gizmos.DrawLine(_back, _isFacingRight ? _leftWallContactPoint : _rightWallContactPoint);
        }

        public void TakeHit(Vector2 hitSourcePosition)
        {
            Debug.Log("Die");
        }
    }
}

        [SerializeField] private float _detectOffsetY;
        [SerializeField] private Transform _muzzle;
        [SerializeField] private BulletPool _bulletPool;

        [SerializeField] private SpriteRenderer _renderer;

        private Animator _animator;
        private Hittable _hittable;

        private float _patrolCenterX;

        private float _idleTimer;
        private EnemyHitter _hitHandler;

        public bool IsFacingLeft
        {
            get { return !_renderer.flipX; }
        }

        private Vector2 FacingDir
        {
            get { return IsFacingLeft ? Vector2.left : Vector2.right; }
        }

        private void Awake()
        {
            _animator = GetComponent<Animator>();
            _renderer = GetComponent<SpriteRenderer>();
            _hittable = GetComponent<Hittable>();
            _hitHandler = new EnemyHitter(_playerMask);

            _animator.SetBool(RunID, true);

            if (_bulletPool == null)
                Debug.LogError($"{nameof(TrunkController)}: BulletPool is not assigned.", this);
        }

        private void Start()
        {
            _patrolCenterX = transform.position.x + _patrolOffsetX;
        }

        private void OnEnable()
        {
            _hittable.OnHit += HandleHit;
        }

        private void OnDisable()
        {
            _hittable.OnHit -= HandleHit;
        }

        private void HandleHit()
        {
            _animator.SetBool(AttackID, false);
            _animator.SetBool(RunID, false);
            _animator.SetTrigger(HitID);
            enabled = false;
        }

        private void Update()
        {
            if (TryGetPlayerInZone(out Collider2D player))
            {
                FaceTowards(player.transform.position.x);
                _animator.SetBool(AttackID, true);
                _animator.SetBool(RunID, false);
                return;
            }

            _animator.SetBool(AttackID, false);

            if (_shouldIdle)
            {
                HandleIdling();
                return;
            }

            Run();
            CheckPatrolEnd();
        }

        private void Run()
        {
            Vector3 direction = IsFacingLeft ? Vector3.left : Vector3.right;
            transform.position += _runSpeed * Time.deltaTime * direction;
            _animator.SetBool(RunID, true);
        }

        private void CheckPatrolEnd()
        {
            float minX = _patrolCenterX - _patrolWidth * 0.5f + _bodyHalfWidth;
            float maxX = _patrolCenterX + _patrolWidth * 0.5f - _bodyHalfWidth;

            Vector3 position = transform.position;

            if (IsFacingLeft && position.x <= minX)
            {
                position.x = minX;
                transform.position = position;
                StartIdling();
            }
            else if (!IsFacingLeft && position.x >= maxX)
            {
                position.x = maxX;
                transform.position = position;
                StartIdling();
            }
        }

        private void StartIdling()
        {
            _shouldIdle = true;
            _idleTimer = _idleTime;

            _animator.SetBool(RunID, false);
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            _hitHandler.TryHitTarget(col, transform.position);
        }

        private void HandleIdling()
        {
            _idleTimer -= Time.deltaTime;

            if (_idleTimer > 0f) return;

            Turn();
            _shouldIdle = false;

        private void Turn()
        {
            _renderer.flipX = !_renderer.flipX;
        }

        private void FaceTowards(float targetX)
        {
            bool shouldFaceLeft = targetX < transform.position.x;

            if (shouldFaceLeft != IsFacingLeft)
                Turn();
        }

        // Detection zone = the patrol area (full patrol width, _detectHeight tall), centered on the patrol center.
        private Vector2 DetectCenter()
        {
            float centerX = Application.isPlaying ? _patrolCenterX : transform.position.x + _patrolOffsetX;
            return new Vector2(centerX, transform.position.y + _detectOffsetY);

        }

        private Vector2 MuzzleOffset()
        {
            if (_muzzle == null)

            return new Vector2(IsFacingLeft ? _muzzle.localPosition.x : -_muzzle.localPosition.x, _muzzle.localPosition.y);

            Vector3 left = new Vector3(center.x - _patrolWidth * 0.5f, y, 0f);

            Gizmos.color = Color.green;

            Gizmos.DrawWireSphere(right, 0.15f);

using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(SpriteRenderer), typeof(Hittable))]
    public class TurtleController : MonoBehaviour
    {
        [SerializeField] private float _idleNoSpikeTime = 1f;
        [SerializeField] private float _idleSpikeTime = 1f;

        [SerializeField] private SingleClipAnimator _idleNoSpike;
        [SerializeField] private SingleClipAnimator _spikeOut;
        [SerializeField] private SingleClipAnimator _idleSpike;
        [SerializeField] private SingleClipAnimator _spikeIn;

        [SerializeField] private SpriteRenderer _renderer;
        [SerializeField] private ContactHazard _hazard = new ContactHazard();
        [SerializeField] private StompImpactTrigger _stompTrigger;

        private Hittable _hittable;
        private SingleClipAnimator _currentClip;

        private void Awake()
        {
            _renderer = GetComponent<SpriteRenderer>();
            _hittable = GetComponent<Hittable>();

            if (_stompTrigger == null)
                _stompTrigger = GetComponent<StompImpactTrigger>();

            SetupClip(_idleNoSpike, true);
            SetupClip(_idleSpike, true);
            SetupClip(_spikeOut, false);
            SetupClip(_spikeIn, false);
        }

        private void OnEnable()
        {
            _hittable.OnHit += HandleHit;
            StartCoroutine(SpikeCycle());
        }

        private void OnDisable()
        {
            _hittable.OnHit -= HandleHit;
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            _hazard.TryHit(col, transform.position);
        }

        private void SetupClip(SingleClipAnimator clip, bool loop)
        {
            clip.monoBehaviour = this;
            clip._renderer = _renderer;
            clip._loop = loop;
        }

        private IEnumerator SpikeCycle()
        {
            while (true)
            {
                SetSpikes(false);
                _hazard.SetIdleNoSpike();
                Play(_idleNoSpike);
                yield return new WaitForSeconds(_idleNoSpikeTime);

                SetSpikes(true);
                bool done = false;
                Play(_spikeOut, () => done = true);
                yield return new WaitUntil(() => done);

                _hazard.SetIdleSpike();
                Play(_idleSpike);
                yield return new WaitForSeconds(_idleSpikeTime);

                _hazard.Disable();
                done = false;
                Play(_spikeIn, () => done = true);
                yield return new WaitUntil(() => done);
            }
        }

        private void Play(SingleClipAnimator clip, System.Action onComplete = null)
        {
            if (_currentClip != null && _currentClip != clip)
                _currentClip.StopClip();

            _currentClip = clip;
            clip.PlayClip(onComplete);
        }

        private void SetSpikes(bool spikesOn)
        {
            _hittable.enabled = !spikesOn;
            _hazard.Disable();

            if (_stompTrigger != null)
                _stompTrigger.enabled = !spikesOn;
        }

        private void HandleHit()
        {
            StopAllCoroutines();

            _currentClip?.StopClip();

            _hazard.Disable();

            if (_stompTrigger != null)
                _stompTrigger.enabled = false;

            enabled = false;
        }
    }
}

using UnityEngine;
using UnityEngine.Pool;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Rigidbody2D), typeof(SpriteRenderer), typeof(Collider2D))]
    public class Bullet : MonoBehaviour
    {
        [SerializeField] private float _speed = 6f;

        [SerializeField] private LayerMask _hitMask;
        [SerializeField]private LayerMask _playerMask;

        [SerializeField] private SpriteRenderer _renderer;
        [SerializeField] private Collider2D _collider;

        [SerializeField] private BulletChunk[] _chunks;
        [SerializeField] private float _chunkBounceSpeed = 2f;
        [SerializeField] private float _chunkUpSpeed = 3f;
        [SerializeField] private float _chunkVariation = 0.3f;
        [SerializeField] private float _chunkSpin = 180f;

        private Rigidbody2D _rb;
        private IObjectPool<Bullet> _pool;

        private Vector2 _direction;
        private bool _isBroken;
        private int _activeChunks;

        private void Awake()
        {
            _rb = GetComponent<Rigidbody2D>();

            _collider = GetComponent<Collider2D>();
            _renderer = GetComponent<SpriteRenderer>();
        }

        private void Start()
        {
            foreach (BulletChunk chunk in _chunks)
            {
                chunk.Initialize();
                chunk.Finished += HandleChunkBlinkEnd;
            }
        }

        private void OnDestroy()
        {
            foreach (BulletChunk chunk in _chunks)
            {
                if (chunk != null)
                    chunk.Finished -= HandleChunkBlinkEnd;
            }
        }

        private void OnTriggerEnter2D(Collider2D col)
        {
            if (!_isBroken && (_playerMask & (1 << col.gameObject.layer)) != 0)
            {
                if(col.TryGetComponent(out IHittable component))
                {
                    component.TakeHit(transform.position);
                    Split();
                    return;
                }
            }

            if (!_isBroken && (_hitMask.value & (1 << col.gameObject.layer)) != 0)
                Split();
        }

        public void SetPool(IObjectPool<Bullet> pool)
        {
            _pool = pool;
        }

        public void Launch(Vector2 direction)
        {
            ResetState();

            _direction = direction.normalized;
            _rb.linearVelocity = _direction * _speed;
        }

        private void Split()
        {
            _isBroken = true;
            _rb.linearVelocity = Vector2.zero;

            _renderer.enabled = false;
            _collider.enabled = false;

            if (_chunks.Length == 0)
            {
                Release();
                return;
            }

            Vector2 back = -_direction;
            _activeChunks = _chunks.Length;

            foreach (BulletChunk chunk in _chunks)
            {
                float backSpeed = _chunkBounceSpeed * RandomVariation();
                float upSpeed = _chunkUpSpeed * RandomVariation();

                Vector2 velocity = back * backSpeed + Vector2.up * upSpeed;
                float spin = _chunkSpin * Random.Range(-1f, 1f);

                chunk.Launch(velocity, spin);
            }
        }

        private void HandleChunkBlinkEnd(BulletChunk chunk)
        {
            if (!_isBroken) return;

            _activeChunks--;

            if (_activeChunks <= 0) Release();
        }

        private void ResetState()
        {
            _isBroken = false;
            _activeChunks = 0;

            _rb.linearVelocity = Vector2.zero;
            _renderer.enabled = true;
            _collider.enabled = true;

            foreach (BulletChunk chunk in _chunks)
                chunk.ResetState();
        }

        private float RandomVariation()
        {
            return 1f + Random.Range(-_chunkVariation, _chunkVariation);
        }

        private void Release()
        {
            if (!gameObject.activeSelf || _pool == null) return;

            _rb.linearVelocity = Vector2.zero;
            _pool.Release(this);
        }
    }
}

using System;
using UnityEngine;
using UnityEngine.Pool;

namespace PixelPlatformer
{
    public class BulletPool : MonoBehaviour
    {
        [SerializeField] private Bullet _bulletPrefab;
        [SerializeField, Min(1)] private int _defaultCapacity = 10;
        [SerializeField, Min(1)] private int _maxSize = 30;

        private ObjectPool<Bullet> _pool;

        private void Awake()
        {
            if (_bulletPrefab == null)
            {
                Debug.LogError($"{nameof(BulletPool)}: Bullet prefab is not assigned.", this);
                return;
            }

            _defaultCapacity = Mathf.Max(1, _defaultCapacity);
            _maxSize = Mathf.Max(_defaultCapacity, _maxSize);

            _pool = new ObjectPool<Bullet>(createFunc: CreateBullet,
                                           actionOnGet: OnGetBullet,
                                           actionOnRelease: OnReleaseBullet,
                                           actionOnDestroy: OnDestroyBullet,
                                           collectionCheck: true,
                                           defaultCapacity: _defaultCapacity,
                                           maxSize: _maxSize);
        }

        public Bullet GetBullet(Vector3 position, Quaternion rotation)
        {
            if (_pool == null)
            {
                Debug.LogError("BulletPool is not initialized.", this);
                return null;
            }

            Bullet bullet = _pool.Get();

            bullet.transform.SetPositionAndRotation(position, rotation);

            return bullet;
        }

        public void ReleaseBullet(Bullet bullet)
        {
            if (_pool == null || bullet == null)
                return;

            _pool.Release(bullet);
        }

        private Bullet CreateBullet()
        {
            Bullet bullet = Instantiate(_bulletPrefab, transform);
            bullet.SetPool(_pool);
            bullet.gameObject.SetActive(false);

            return bullet;
        }

        private void OnGetBullet(Bullet bullet)
        {
            bullet.gameObject.SetActive(true);
        }

        private void OnReleaseBullet(Bullet bullet)
        {
            bullet.gameObject.SetActive(false);
            bullet.transform.SetParent(transform);
        }

        private void OnDestroyBullet(Bullet bullet)
        {
            if (bullet != null)
                Destroy(bullet.gameObject);
        }
    }

using System;
using UnityEngine;

namespace PixelPlatformer
{
    [Flags]
    public enum Direction
    {
        None = 0,
        Up = 1 << 0,
        Down = 1 << 1,
        Right = 1 << 2,
        Left = 1 << 3,
    }

    [Serializable]
    public class ContactHazard
    {
        [SerializeField] private LayerMask _targetMask;
        [SerializeField] private Direction _idleSpikeDirections;
        [SerializeField] private Direction _idleNoSpikeDirections;

        private Direction _activeDirections;

        public Direction ActiveDirections
        {
            get
            {
                return _activeDirections;
            }
        }

        public void SetIdleSpike()
        {
            _activeDirections = _idleSpikeDirections;
        }

        public void SetIdleNoSpike()
        {
            _activeDirections = _idleNoSpikeDirections;
        }

        public void Disable()
        {
            _activeDirections = Direction.None;
        }

        public void TryHit(Collision2D col, Vector2 hitPosition)
        {
            if (_activeDirections == Direction.None) return;

            if ((_targetMask.value & (1 << col.gameObject.layer)) == 0) return;

            if (!col.collider.TryGetComponent<IHittable>(out var hittable) || !hittable.IsHittable)
                return;

            for (int i = 0; i < col.contactCount; i++)
            {
                Vector2 toTarget = -col.GetContact(i).normal;

                if ((_activeDirections & GetSide(toTarget)) != 0)
                {
                    hittable.TakeHit(hitPosition);
                    return;
                }
            }
        }

        private static Direction GetSide(Vector2 toTarget)
        {
            if (Mathf.Abs(toTarget.y) >= Mathf.Abs(toTarget.x))
                return toTarget.y > 0f ? Direction.Up : Direction.Down;

            return toTarget.x > 0f ? Direction.Right : Direction.Left;
        }

        public void SetAllDirections()
        {
            _activeDirections = Direction.Up | Direction.Down | Direction.Left | Direction.Right;
        }
    }
}

using UnityEngine;

namespace PixelPlatformer
{
    public class EnemyHitter
    {
        private readonly LayerMask _targetMask;
        private readonly float _threshold;

        public EnemyHitter(LayerMask targetMask, float threshold = 0.5f)
        {
            _targetMask = targetMask;
            _threshold = threshold;
        }

        public void TryHitTarget(Collision2D col, Vector3 hitPosition)
        {
            if ((_targetMask.value & (1 << col.gameObject.layer)) == 0)
                return;

            if (!col.collider.TryGetComponent<IHittable>(out var hittable)
                || !hittable.IsHittable)
                return;

            // direction from us toward the target
            Vector2 toTarget = -col.GetContact(0).normal;

            // Ignore the target if it's on top
            if (toTarget.y > _threshold)
                return;

            hittable.TakeHit(hitPosition);
        }
    }
}

using System;
using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Collider2D))]
    public class Hittable : MonoBehaviour, IHittable
    {
        [SerializeField] private float _turnAngle = 15f;
        [SerializeField] private float _jumpHeight = 0.8f;
        [SerializeField] private float _jumpDuration = 0.6f;
        [SerializeField] private float _turnSpeed = 0.2f;
        [SerializeField] private float _fallAcceleration = 1f;
        [SerializeField] private float _fallSpeed = 0.5f;
        [SerializeField] private SpriteRenderer _renderer;
        private Collider2D _collider;

        private bool _isHittable = true;
        private float _currentFallSpeed;
        private int _defaultSortingOrder;

        public event Action OnHit;
        public event Action OnDied;

        public bool IsHittable { get { return _isHittable && enabled; } }

        private void Awake()
        {
            _collider = GetComponent<Collider2D>();
            _defaultSortingOrder = _renderer.sortingOrder;
        }

        public void ResetHit()
        {
            StopAllCoroutines();
            _currentFallSpeed = 0f;
            _collider.enabled = true;
            _renderer.sortingOrder = _defaultSortingOrder;
        }

        public void TakeHit(Vector2 hitSourcePosition)
        {
            GameEvents.Publish(new CameraShakeEventData());

            _collider.enabled = false;
            _renderer.sortingOrder = 50;

            StartCoroutine(HitRoutine());

            OnHit?.Invoke();
        }

        private IEnumerator HitRoutine()
        {
            Vector2 startPos = transform.position;
            float angle = UnityEngine.Random.Range(-_turnAngle, _turnAngle);

            float elapsedTime = 0f;

            // jump
            while (elapsedTime <= _jumpDuration)
            {
                elapsedTime += Time.deltaTime; // update elapsed time each frame;

                float progress = Mathf.Clamp01(elapsedTime / _jumpDuration); // progress goes from 0 to 1

                float easeOut = 1f - Mathf.Pow(1f - progress, 2); // see https://notes.yvt.jp/Graphics/Easing-Functions/

                Vector2 pos = startPos;
                pos.y += _jumpHeight * easeOut;

                transform.SetPositionAndRotation(pos, Quaternion.Slerp(transform.rotation, Quaternion.Euler(Vector3.forward * angle), _turnSpeed));

                yield return null;
            }

            // fall slowly using acceleration 
            while (gameObject.activeSelf)
            {
                Vector2 pos = transform.position;

                _currentFallSpeed = Mathf.MoveTowards(_currentFallSpeed, _fallSpeed, _fallAcceleration * Time.deltaTime);
                pos.y -= _currentFallSpeed;

                transform.position = pos;

                // viewport co-ordinate origin is (0,0) which is bottom left corner and top right corner is (1,1)
                // convert world position to viewport co-ordinate and then check if that value is less than 0 which is viewport bottom bound
                if (Camera.main.WorldToViewportPoint(pos).y < 0)
                {
                    OnDied?.Invoke();
                    gameObject.SetActive(false);
                }
                yield return null;
            }
        }
    }

using System.Collections;
using UnityEngine;
using UnityEngine.Pool;

namespace PixelPlatformer
{
    public class SlimeParticle : MonoBehaviour
    {
        [SerializeField] private LayerMask _targetMask;
        [SerializeField] private SingleClipAnimator _animator;
        [SerializeField] private float _playDelay = 2f;

        private IObjectPool<SlimeParticle> _pool;
        private Coroutine _delayRoutine;

        private void Awake()
        {
            _animator.monoBehaviour = this;

            if (_animator._renderer == null)
                _animator._renderer = GetComponent<SpriteRenderer>();
        }

        private void OnDisable()
        {
            StopDelay();
        }

        public void SetPool(IObjectPool<SlimeParticle> pool)
        {
            _pool = pool;
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            if ((_targetMask.value & (1 << col.gameObject.layer)) == 0) return;

            if (!col.gameObject.TryGetComponent<IHittable>(out var hittable) || !hittable.IsHittable)
                return;

            hittable.TakeHit(transform.position);
        }

        public void PlayClip()
        {
            StopDelay();

            if (_animator._sprites != null && _animator._sprites.Length > 0)
                _animator._renderer.sprite = _animator._sprites[0];

            _delayRoutine = StartCoroutine(PlayAfterDelay());
        }

        private IEnumerator PlayAfterDelay()
        {
            yield return new WaitForSeconds(_playDelay);

            _delayRoutine = null;
            _animator.PlayClip(Release);
        }

        private void StopDelay()
        {
            if (_delayRoutine == null) return;

            StopCoroutine(_delayRoutine);
            _delayRoutine = null;
        }

        private void Release()
        {
            if (!gameObject.activeSelf || _pool == null) return;

            StopDelay();
            _animator.StopClip();
            _pool.Release(this);
        }
    }
}

using System;
using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Rigidbody2D))]
    public class ContainerChunk : MonoBehaviour
    {
        [SerializeField] private int _totalKeyframes = 12;
        [SerializeField] private LayerMask _groundLayer;
        public int _frameRate = 12;
        public SpriteRenderer _renderer;
        public Sprite _sprite;
        private Rigidbody2D _rb;
        private int _keyFrame;

        private void Awake()
        {
            _rb = GetComponent<Rigidbody2D>();
        }

        private void OnCollisionEnter2D(Collision2D other)
        {
            if ((_groundLayer & (1 << other.gameObject.layer)) != 0)
            {
                StartCoroutine(PlayClipRoutine());
            }
        }

        public void ApplyImpulseForce(Vector2 force)
        {
            _rb.AddForce(force, ForceMode2D.Impulse);
        }

        private IEnumerator PlayClipRoutine()
        {
            _keyFrame = 0;
            var elapsedTime = 0f;
            var targetDeltaTime = 1f / _frameRate;

            // for every 1/12the time i.e 12fps considering default framerate, jump to next keyframe
            while (_keyFrame < _totalKeyframes)
            {
                elapsedTime += Time.deltaTime;

                if (elapsedTime >= targetDeltaTime)
                {
                    Sprite sprite = _keyFrame % 2 == 0 ? null : _sprite;
                    _renderer.sprite = sprite;
                    _keyFrame++;
                    elapsedTime -= targetDeltaTime;
                }

                yield return null;
            }

            gameObject.SetActive(false);
        }
    }
}

using UnityEngine;
using UnityEngine.Serialization;

namespace PixelPlatformer
{
    public interface IBreakable
    {
        bool IsBreakable { get; }
        void Break();
    }

    public class FruitsContainer : MonoBehaviour, IBreakable
    {
        [SerializeField] private ContainerChunk[] _chunks;
        [SerializeField] private DynamicFruit[] _dynamicFruits;
        [SerializeField] private int _totalHits = 1;
        [SerializeField] private SingleClipAnimator _hitClipAnimator;

        [Header("Spread Settings")]
        [FormerlySerializedAs("_minImpulseForce")]
        [SerializeField] private float _baseImpulseForce = 4f;
        [SerializeField] private float _forceIncrement = 1f;

        private int _remainingHits;
        private bool _isBreakable = true;

        public bool IsBreakable { get { return _isBreakable; } }

        private void Awake()
        {
            _remainingHits = _totalHits;
        }

        private void Start()
        {
            foreach (ContainerChunk chunk in _chunks)
            {
                chunk.gameObject.SetActive(false);
            }

            foreach (DynamicFruit fruit in _dynamicFruits)
            {
                fruit.gameObject.SetActive(false);
            }
        }

        public void Break()
        {
            _hitClipAnimator.PlayClip();
            _remainingHits--;

            if (_remainingHits <= 0)
            {
                _isBreakable = false;
                GetComponent<Collider2D>().enabled = false;

                SpawnChunks();
            }
        }

        private void SpawnChunks()
        {
            _hitClipAnimator._renderer.enabled = false;

            int rightCount = 0;
            int leftCount = 0;

            for (int i = 0; i < _chunks.Length; i++)
            {
                _chunks[i].gameObject.SetActive(true);
                _chunks[i].ApplyImpulseForce(GetSideForce(i % 2 == 0, ref rightCount, ref leftCount));
            }

            rightCount = 0;
            leftCount = 0;

            for (int i = 0; i < _dynamicFruits.Length; i++)
            {
                _dynamicFruits[i].gameObject.SetActive(true);
                _dynamicFruits[i].Throw(GetSideForce(i % 2 == 0, ref rightCount, ref leftCount));
            }
        }

        private Vector2 GetSideForce(bool isRight, ref int rightCount, ref int leftCount)
        {
            int count = isRight ? rightCount++ : leftCount++;

            Vector2 direction = isRight ? Vector2.right : Vector2.left;
            float force = _baseImpulseForce + count * _forceIncrement;

            return direction * force;
        }
    }
}

using UnityEngine;

namespace PixelPlatformer
{
    [System.Serializable]
    public struct LevelObjectives
    {
        public int totalFruits;
        public int totalEnemies;
        public float estimatedCompletionTime;
    }
    public class LevelManager : Singleton<LevelManager>
    {
        [SerializeField] private float _despawnHeight = 2.5f;
        [SerializeField] private LevelObjectives _levelObjectives;
        [SerializeField] private LevelScoreDataSO _scoreData;

        [SerializeField] private StartCheckpoint _startCheckpoint;
        [SerializeField] private EndCheckpoint _endCheckpoint;
        [SerializeField] private PlayerSpawnFX _playerSpawnFX;
        [SerializeField] private PlayerDespawnFX _playerDespawnFX;
        [SerializeField] private AnimatedBackground _background;

        private UIManager _uIManager;
        private GameplayManager _gameplayManager;
        private PlayerManager _playerManager;
        private LevelRecord _record;
        private float _maxScore;
        private int _enemiesDefeated;
        private int _fruitsCollected;
        private bool _hasCompleted;

        private void OnEnable()
        {
            _startCheckpoint.OnPlayerLand += HandlePlayerSpawned;
            _endCheckpoint.OnTrigger += HandleLevelCompleted;

            GameEvents.Subscribe<FruitCollectedEvent>(HandleFruitCollected);
            GameEvents.Subscribe<StompedEnemyEventData>(HandleStomp);
        }

        private void OnDisable()
        {
            _startCheckpoint.OnPlayerLand -= HandlePlayerSpawned;
            _endCheckpoint.OnTrigger -= HandleLevelCompleted;

            GameEvents.Unsubscribe<FruitCollectedEvent>(HandleFruitCollected);
            GameEvents.Unsubscribe<StompedEnemyEventData>(HandleStomp);
        }

        private void OnDestroy()
        {
            _playerManager.OnFirstMove -= HandleMoveMade;
        }

        private void HandleMoveMade()
        {
            _gameplayManager.StartTimer();
        }

        public void SetupLevel(LevelRecord record)
        {
            _record = record;

            _hasCompleted = false;

            _enemiesDefeated = 0;
            _fruitsCollected = 0;

            _maxScore = ScoreCalculator.MaxScore(_levelObjectives, _scoreData);

            _uIManager = UIManager.Instance;
            _gameplayManager = GameplayManager.Instance;
            _playerManager = PlayerManager.Instance;

            // reset HUD
            _uIManager.ResetHUD();

            // prepare the win screen for this level
            _uIManager.SetupWinScreen(GameManager.Instance.GetLevelIcon(record.levelIndex),
                                      _levelObjectives.totalEnemies,
                                      _levelObjectives.totalFruits);

            // avoid subscribing to the same event more than once
            _playerManager.OnFirstMove -= HandleMoveMade;
            _playerManager.OnFirstMove += HandleMoveMade;

            _playerManager.SetupPlayer(_playerSpawnFX.transform.position);

            _background.Setup(_playerManager.FollowCam);
        }

        public void StartLevel()
        {
            _playerSpawnFX.PlayFX(() =>
            {
                _playerManager.SpawnPlayer();
            });
        }

        public void GoToNextLevel()
        {
            _playerDespawnFX.transform.position =
            _playerManager.PlayerPosition +
            Vector3.up * _despawnHeight;

            _playerManager.DespawnPlayerAt(_playerDespawnFX.transform.position, () =>
            {
                _playerDespawnFX.PlayFX(() =>
                {
                    GameManager.Instance.GoToNextLevel();
                });
            });
        }

        private void HandlePlayerSpawned()
        {
            if (_gameplayManager.State != GameplayState.Play)
                return;

            _playerManager.ToggleInput(true);
        }

        private void HandleLevelCompleted()
        {
            if (_hasCompleted || _gameplayManager.State != GameplayState.Play)
                return;

            _hasCompleted = true;

            _playerManager.ToggleInput(false);
            _gameplayManager.StopTimer();

            LevelResult result = new LevelResult
            {
                timeTaken = _gameplayManager.ElapsedTime,
                fruitsCollected = _fruitsCollected,
                enemiesDefeated = _enemiesDefeated,
                totalFruits = _levelObjectives.totalFruits,
                totalEnemies = _levelObjectives.totalEnemies
            };

            result.score = Mathf.Max(0f, Mathf.Round(ScoreCalculator.TotalScore(result,
                                                                                _scoreData,
                                                                                _levelObjectives.estimatedCompletionTime)));

            result.starsEarned = ScoreCalculator.GetStars(result.score, _maxScore, 3);

            // read the previous best before the event saves the new record
            float previousBest = _record != null ? _record.bestScore : 0f;

            result.isNewBest = result.score > previousBest;

            result.bestScore = Mathf.Max(previousBest, result.score);

            GameEvents.Publish(new LevelCompletedEvent(result));
        }

        private void HandleStomp(StompedEnemyEventData data)
        {
            _enemiesDefeated++;

            _uIManager.SetEnemiesDefeated(_enemiesDefeated);
        }

        private void HandleFruitCollected(FruitCollectedEvent data)
        {
            _fruitsCollected++;

            _uIManager.SetCollectibles(_fruitsCollected);
        }
    }
}

// ── FILE: ScoreCalculator.cs ──

using UnityEngine;

namespace PixelPlatformer
{
    public static class ScoreCalculator
    {
        public static int GetStars(float score, float maxScore, int maxStars)
        {
            float ratio = Mathf.Clamp01(score / maxScore);

            if (ratio >= 0.85f) return maxStars;
            if (ratio >= 0.5f) return Mathf.Max(1, maxStars - 1);

            return 1;
        }

        public static float TotalScore(LevelResult result, LevelScoreDataSO levelScore, float estimatedCompletionTime)
        {
            float timeBonus = (estimatedCompletionTime * levelScore.timeBonus) + (estimatedCompletionTime - result.timeTaken) * levelScore.timePoint;

            return (result.fruitsCollected * levelScore.fruitPoint)
                         + (result.enemiesDefeated * levelScore.enemyPoint)
                         + timeBonus;
        }

        public static float MaxScore(LevelObjectives objectives, LevelScoreDataSO levelScore)
        {
            return (objectives.totalFruits * levelScore.fruitPoint)
                   + (objectives.totalEnemies * levelScore.enemyPoint)
                   + (objectives.estimatedCompletionTime * levelScore.timeBonus);
        }
    }
}

using UnityEngine;

namespace PixelPlatformer
{
    public class PathFollower : MonoBehaviour
    {
        [SerializeField] private PathLine _path;
        [SerializeField] private float _speed = 2f;
        [SerializeField] private bool _snapToStart = true;

        private float _distance;
        private int _direction = 1;

        private void Start()
        {
            if (_path == null) return;

            if (_snapToStart)
                SetPosition(_path.GetPosition(0f));
        }

        private void Update()
        {
            if (_path == null || _path.Length <= 0f) return;

            _distance += _direction * _speed * Time.deltaTime;

            if (_path.IsClosed)
            {
                _distance = Mathf.Repeat(_distance, _path.Length);
            }
            else if (_distance >= _path.Length)
            {
                _distance = _path.Length;
                _direction = -1;
            }
            else if (_distance <= 0f)
            {
                _distance = 0f;
                _direction = 1;
            }

            SetPosition(_path.GetPosition(_distance));
        }

        private void SetPosition(Vector2 position)
        {
            transform.position = new Vector3(position.x, position.y, transform.position.z);
        }
    }

// ── FILE: PathLine.cs ──
// Path: Assets/_Platformer/_Scripts/Path/PathLine.cs

using UnityEngine;
#if UNITY_EDITOR
using UnityEditor;
using UnityEditor.SceneManagement;
#endif

namespace PixelPlatformer
{
    public class PathLine : MonoBehaviour
    {
        [SerializeField] private Transform[] _points;
        [SerializeField] private bool _closed;

        [SerializeField] private Transform _pathRoot;
        [SerializeField] private GameObject _chainPrefab;
        [SerializeField] private float _chainSpacing = 0.5f;

        private float[] _cumulative;
        private float _length;

        public bool IsClosed
        {
            get { return _closed; }
        }

        public float Length
        {
            get { return _length; }
        }

        private void Awake()
        {
            Build();
        }

        private void Build()
        {
            _length = 0f;

            if (_points == null || _points.Length < 2)
            {
                _cumulative = new float[0];
                return;
            }

            int segments = SegmentCount();
            _cumulative = new float[segments];

            for (int i = 0; i < segments; i++)
            {
                _length += Vector2.Distance(PointAt(i), PointAt(i + 1));
                _cumulative[i] = _length;
            }
        }

        public Vector2 GetPosition(float distance)
        {
            if (_length <= 0f)
                return _points != null && _points.Length > 0 ? (Vector2)_points[0].position : (Vector2)transform.position;

            distance = _closed ? Mathf.Repeat(distance, _length) : Mathf.Clamp(distance, 0f, _length);

            float segmentStart = 0f;

            for (int i = 0; i < _cumulative.Length; i++)
            {
                if (distance <= _cumulative[i])
                {
                    float segmentLength = _cumulative[i] - segmentStart;
                    float t = segmentLength > 0f ? (distance - segmentStart) / segmentLength : 0f;

                    return Vector2.Lerp(PointAt(i), PointAt(i + 1), t);
                }

                segmentStart = _cumulative[i];
            }

            return PointAt(_points.Length - 1);
        }

        private int SegmentCount()
        {
            return _closed ? _points.Length : _points.Length - 1;
        }

        private Vector2 PointAt(int index)
        {
            return _points[index % _points.Length].position;
        }

        private void OnDrawGizmos()
        {
            if (_points == null || _points.Length == 0) return;

            Gizmos.color = Color.green;

            for (int i = 0; i < _points.Length; i++)
            {
                if (_points[i] == null) continue;

                Gizmos.DrawWireSphere(_points[i].position, 0.15f);

                bool hasNext = i < _points.Length - 1 || (_closed && _points.Length > 2);

                if (!hasNext) continue;

                Transform next = _points[(i + 1) % _points.Length];

                if (next != null)
                    Gizmos.DrawLine(_points[i].position, next.position);
            }
        }

#if UNITY_EDITOR
        private void OnValidate()
        {
            _chainSpacing = Mathf.Max(0.1f, _chainSpacing);

            if (Application.isPlaying || EditorUtility.IsPersistent(this)) return;

            EditorApplication.delayCall -= RebuildChain;
            EditorApplication.delayCall += RebuildChain;
        }

        [ContextMenu("Rebuild Chain")]
        private void RebuildChain()
        {
            if (this == null || _pathRoot == null || _chainPrefab == null) return;
            if (_points == null || _points.Length < 2) return;

            for (int i = 0; i < _points.Length; i++)
            {
                if (_points[i] == null) return;
            }

            Build();

            for (int i = _pathRoot.childCount - 1; i >= 0; i--)
                DestroyImmediate(_pathRoot.GetChild(i).gameObject);

            if (_length <= 0f) return;

            int segments = Mathf.Max(1, Mathf.RoundToInt(_length / _chainSpacing));
            float step = _length / segments;
            int count = _closed ? segments : segments + 1;

            for (int i = 0; i < count; i++)
            {
                GameObject link = PrefabUtility.InstantiatePrefab(_chainPrefab, _pathRoot) as GameObject;
                if (link == null) break;

                Vector2 position = GetPosition(i * step);
                link.transform.position = new Vector3(position.x, position.y, _pathRoot.position.z);
            }

            EditorSceneManager.MarkSceneDirty(gameObject.scene);
        }
#endif
    }
}

            TryHit(col);
        }

        private void OnTriggerStay2D(Collider2D col)
        {
            TryHit(col);
        }

        private void TryHit(Collider2D col)
        {
            if ((_targetMask.value & (1 << col.gameObject.layer)) == 0) return;

            if (!col.TryGetComponent<IHittable>(out var hittable) || !hittable.IsHittable)
                return;

            hittable.TakeHit(transform.position);
        }
    }
}

        public event Action OnJumpCanceled;

            _mobileInputUIRoot.SetActive(_useMobileInput);

                ReadMobileInput();

                ReadKeyboardInput();

                ReadMobileInput();

// ── FILE: PlayerMovement.cs ──
// Path: Assets/_Platformer/_Scripts/Player/Movement/PlayerMovement.cs

using System;
using UnityEngine;
#if UNITY_EDITOR
using GP.Utils.Gizmo;
#endif

namespace PixelPlatformer
{
    /// <summary>
    /// Component responsible for rigidbody based movement of the player
    /// </summary>
    [RequireComponent(typeof(Rigidbody2D))]
    public class PlayerMovement : MonoBehaviour
    {
        private const float MOVEMENT_THRESHOLD = 0.01f;
        private const float VERTICAL_NORMAL_THRESHOLD = 0.5f;


        [SerializeField] private PlayerMovementDataSO _movementStats;

        [Header("Component References")]
        [SerializeField] private Transform _groundCheckPoint;
        [SerializeField] private Transform _ceilingCheckPoint;
        [SerializeField] private Transform _wallCheckPoint;
        [SerializeField] private Rigidbody2D _rb;

#if UNITY_EDITOR
        [Header("Debugging")]
        [SerializeField] private bool _toggleGizmos;
        [SerializeField] private bool _showDebugLabels;
        [SerializeField] private float _lineThickness = 5f;
        [SerializeField] private Color _verticalArrowColor = Color.green;
        [SerializeField] private Color _horizontalArrowColor = Color.yellow;
        [SerializeField] private float _arrowWidth = 1f;
        [SerializeField] private float _arrowHeight = 1f;
        [SerializeField] private Vector3 _offset = Vector2.down;
#endif


        // private fields
        private float _moveInput;

        private float _gravity;
        private LayerMask _walkableMask;

        private float _initialJumpVelocity; // velocity required to reach jump apex
        private float _verticalVelocity;
        private float _horizontalVelocity;

        private bool _wasGrounded; // ground flag for previous frame
        private float _jumpBufferTimer;
        private float _coyoteTimer;

        private bool _wasFacingRight;
        private bool _wasJumping;

        private bool _isWallSliding;
        private float _wallJumpLockTimer;
        private bool _isFlying;


        // Properties
        public bool IsRunning
        {
            get
            {
                return IsGrounded() && Mathf.Abs(_horizontalVelocity) > MOVEMENT_THRESHOLD;
            }
        }

        public int LookDirection { get; private set; }
        public Vector3 LinearVelocity { get { return _rb.linearVelocity; } }

        public bool IsFlying { get { return _isFlying; } }


        // events
        public event Action OnMove;
        public event Action<bool> OnTurn; // true means facing right
        public event Action OnFall;

        public event Action OnLand;
        public event Action OnJump;
        public event Action<bool> OnWallSlide;





        #region Unity API Methods
        private void Awake()
        {
            SetupJumpVariables();
        }

        private void OnEnable()
        {
            SetWalkableMask(_movementStats.GroundMask);
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            // order matters since onw way becomes ground when player is right above the one way
            HandleOneWayGroundMovement(col);
            CaptureFallVelocityOnLanding(col);
        }

        private void OnCollisionExit2D(Collision2D col)
        {
            // remove the one way from walkable after the player exits one way
            RemoveOnewayFromWalkableMask(col);
        }
        #endregion







        #region One Way Handling Methods
        // while falling set the vertical velocity to rigidbody's vertical velocity, 
        // otherwise vertical velocity reaches high and player will slam the ground
        private void CaptureFallVelocityOnLanding(Collision2D col)
        {
            if ((_walkableMask & (1 << col.gameObject.layer)) != 0 && IsFalling())
            {
                _verticalVelocity = _rb.linearVelocityY;
            }
        }

        // include the one way when the player is exactly above the platform
        private void HandleOneWayGroundMovement(Collision2D col)
        {
            // check if contact has made
            if ((_movementStats.OneWayMask & (1 << col.gameObject.layer)) != 0)
            {
                for (int i = 0; i < col.contactCount; i++)
                {
                    ContactPoint2D point = col.GetContact(i);

                    if (point.normal.y > VERTICAL_NORMAL_THRESHOLD) // normal y is 1 means flat, 0.5 means 45 degree slop
                    {
                        SetWalkableMask(_movementStats.GroundMask | _movementStats.OneWayMask.value);
                        return;
                    }
                }
            }
        }

        private void RemoveOnewayFromWalkableMask(Collision2D col)
        {
            // set the walkable mask to ground mask
            if ((_movementStats.OneWayMask & (1 << col.gameObject.layer)) != 0)
                SetWalkableMask(_movementStats.GroundMask);
        }

        private void SetWalkableMask(LayerMask mask)
        {
            _walkableMask = mask;
        }
        #endregion

        #region Movement Methods
        /// <summary>
        /// Movements are based on player input 
        /// </summary>
        /// <param name="moveInput"></param>
        public void SimulateMovement(float moveInput)
        {
            _moveInput = moveInput;

            HandleWallSlide();
            HandleJump();
            HandleHorizontalMovement();
            ApplyGravity();
        }

        /// <summary>
        /// Straight line movement
        /// </summary>
        /// <param name="target"></param>
        /// <returns></returns>
        public bool StepTowards(Vector3 target, float speed)
        {
            if ((transform.position - target).magnitude <= 0.01f)
            {
                _rb.MovePosition(target);
                return true;
            }

            _rb.MovePosition(Vector3.MoveTowards(transform.position, target, speed));
            return false;
        }
        private void HandleHorizontalMovement()
        {
            if (_wallJumpLockTimer > 0f)
            {
                _wallJumpLockTimer -= Time.fixedDeltaTime;
                return;
            }

            // The target speed will be max speed when we press input otherwise it will be zero
            float targetSpeed = _moveInput * _movementStats.MoveSpeed;
            float speedChange;

            if (Mathf.Abs(targetSpeed) > MOVEMENT_THRESHOLD)
            {
                // If the sign (i.e. positive or negative) of our input direction doesn't match our movement, it means we're turning around
                if (Mathf.Sign(targetSpeed) != Mathf.Sign(_horizontalVelocity))
                    speedChange = _movementStats.TurnSpeed;
                else
                    speedChange = IsGrounded() ? _movementStats.GroundAcceleration : _movementStats.AirAcceleration;
            }
            else
            {
                speedChange = IsGrounded() ? _movementStats.GroundDecceleration : _movementStats.AirDecceleration;
            }

            // Mathf.MoveTowards uses constant speedchange
            _horizontalVelocity = Mathf.MoveTowards(_horizontalVelocity, targetSpeed, speedChange * Time.fixedDeltaTime);

            if (Mathf.Abs(_moveInput) > MOVEMENT_THRESHOLD)
            {
                bool facingRight = _moveInput > 0f;

                if (facingRight != _wasFacingRight)
                {
                    _wasFacingRight = facingRight;
                    LookDirection = facingRight ? 1 : -1;
                    OnTurn?.Invoke(facingRight);
                }
                OnMove?.Invoke();
            }
        }

        public void ApplyMovement()
        {
            _rb.linearVelocity = new Vector2(_horizontalVelocity, _verticalVelocity);
        }
        #endregion

        #region Initialization Methods
        public void ResetState()
        {
            ResetVerticalVelocity();
            _horizontalVelocity = 0f;
            _isWallSliding = false;
            _wasJumping = false;
            _wallJumpLockTimer = 0f;
            _jumpBufferTimer = 0f;
            _coyoteTimer = 0f;

            // physics state
            _rb.linearVelocity = Vector2.zero;
            _rb.angularVelocity = 0f;
            _rb.rotation = 0f;
            transform.rotation = Quaternion.identity;
        }

        // calculate gravity using parabola eq and set the initial velocity for jump
        private void SetupJumpVariables()
        {
            /*position equation p(t) = p0 + v0*t + (1/2)*g*t^2
                      We only care about height gained, so set p0 = 0
                      then eq becomes h(t) = v0*t + (1/2)*g*t^2
                      At the apex (t = timeTillJumpApex), vertical velocity is 0: 
                      after differentiating we get v(t) = v0 + g*t = 0 v(t) velocity at apex is zero then v0 = -g*t
                      Substituting v0 = -g*t into h(t) eliminates the v0*t term: h = -g*t^2 + (1/2)*g*t^2 = -(1/2)*g*t^2
                      then g = -2h / t^2
                    */
            _gravity = -(2f * _movementStats.JumpHeight) / Mathf.Pow(_movementStats.TimeTillJumpApex, 2) * _movementStats.GravityMuliplier;

            /* From v = g * t: the launch speed needed to reach 0 velocity (the apex)
              after timeTillJumpApex seconds under that gravity
            */
            _initialJumpVelocity = Mathf.Abs(_gravity) * _movementStats.TimeTillJumpApex;
        }
        #endregion


        #region Collision Check
        private bool IsGrounded()
        {
            var hit = Physics2D.BoxCast(_groundCheckPoint.position,
                                        _movementStats.GroundCheckSize,
                                        0f,
                                        Vector2.down,
                                        _movementStats.GroundCheckDistance,
                                        _walkableMask);

            return hit & hit.point.y <= _groundCheckPoint.position.y;
        }

        private bool HitCeiling()
        {
            int mask = _movementStats.GroundMask & ~(1 << _movementStats.OneWayMask); // ignore One Way Platform which is not ceiling

            return Physics2D.BoxCast(_ceilingCheckPoint.position,
                                     _movementStats.CeilingCheckSize,
                                     0f,
                                     Vector2.up,
                                     _movementStats.CeilingCheckDistance,
                                     mask);
        }
        #endregion

        #region Jump
        public void BufferJump()
        {
            // restart timer
            _jumpBufferTimer = _movementStats.JumpBufferTime;
        }

        public void CutJump()
        {
            // Variable jump height
            if (_verticalVelocity > 0f)
                _verticalVelocity *= _movementStats.JumpCutMultiplier;
        }

        private void HandleJump()
        {
            bool isJumpPressed = _jumpBufferTimer > 0;

            // previous frame was airborne, current frame is grounded
            if (!_wasGrounded && IsGrounded())
            {
                OnLand?.Invoke();
            }

            // Jump Input Buffer
            if (!IsGrounded())
            {
                _jumpBufferTimer -= Time.fixedDeltaTime;
            }
            else if (isJumpPressed)
            {
                PerformJump(_initialJumpVelocity);
                return;
            }

            // Wall Jump — jump pressed while sliding, before coyote time applies
            if (_isWallSliding && isJumpPressed)
            {
                PerformWallJump();
                return;
            }

            // Coyote Time
            // check for player walking off edge
            if (_wasGrounded && !IsGrounded())
            {
                // restart timer
                _coyoteTimer = _movementStats.CoyoteTime;
            }

            if (_wasGrounded && IsFalling() && !_wasJumping)
                OnFall?.Invoke();

            // countdown timer
            if (!IsGrounded())
            {
                _coyoteTimer -= Time.fixedDeltaTime;

                if (_coyoteTimer > 0 && isJumpPressed) // _jumpBufferTimer > 0 means jump pressed 
                                                       // or use _jumpPressed flag
                {
                    PerformJump(_initialJumpVelocity);
                }
            }

            _wasGrounded = IsGrounded();
        }

        private void ResetJumpBuffer()
        {
            _jumpBufferTimer = 0; // Set the value to zero not max buffer time otherwise player will keep jumping
        }

        private bool IsJumping()
        {
            return !IsGrounded() && _verticalVelocity > 0f;
        }

        private bool IsFalling()
        {
            return !IsGrounded() && _verticalVelocity <= 0f;
        }

        public void PerformJump(float jumpVelocity)
        {
            _verticalVelocity = jumpVelocity;

            ResetJumpBuffer();
            OnJump?.Invoke();
        }

        private void PerformWallJump()
        {
            float wallDirection = Mathf.Sign(_moveInput);

            _horizontalVelocity = -wallDirection * _movementStats.WallJumpHorizontalForce;
            _verticalVelocity = _movementStats.WallJumpVerticalVelocity;

            // set the wall jump lock time to desired lock time
            _wallJumpLockTimer = _movementStats.WallJumpLockTime;

            _isWallSliding = false;
            OnWallSlide?.Invoke(false);

            ResetJumpBuffer();
            OnJump?.Invoke();
        }
        #endregion


        #region Gravity Methods
        private void ApplyGravity()
        {
            if (IsGrounded() && _verticalVelocity <= 0f)
            {
                ApplyGroundedGravity();
                return;
            }

            if (HitCeiling())
                _verticalVelocity = 0;

            ApplyAirborneGravity();

            if (_isWallSliding)
            {
                _verticalVelocity = Mathf.Max(_verticalVelocity, -_movementStats.WallSlideSpeed);
            }

            // check if previously going up and now coming down then isFalling
            if (_wasJumping && IsFalling())
                OnFall?.Invoke();

            _wasJumping = IsJumping();
        }

        private void ApplyGroundedGravity()
        {
            _verticalVelocity = -_movementStats.GroundedVerticalVelocity;
        }

        private void ApplyAirborneGravity()
        {
            _verticalVelocity += _gravity * Time.fixedDeltaTime;
        }
        #endregion


        #region Wall Slide
        private void HandleWallSlide()
        {
            bool wasWallSliding = _isWallSliding;

            bool hasInput = Mathf.Abs(_moveInput) > MOVEMENT_THRESHOLD;
            bool isFalling = !IsGrounded() && _verticalVelocity <= 0f;

            // must be airborne, falling, and pressing movement causing contact with wall
            _isWallSliding = hasInput && isFalling && IsWallSliding();

            if (_isWallSliding != wasWallSliding)
                OnWallSlide?.Invoke(_isWallSliding);
        }

        private bool IsWallSliding()
        {
            return Physics2D.BoxCast(_wallCheckPoint.position,
                                     _movementStats.WallCheckSize,
                                     0f,
                                     Vector2.right * Mathf.Sign(_moveInput),
                                     _movementStats.WallCheckDistance,
                                     _movementStats.WallMask);
        }
        #endregion

        // switches control of velocity and gravity to manual or unity's physics
        public void SwitchControlType(bool canControl)
        {
            if (canControl)
            {
                _isFlying = false;
                _rb.gravityScale = 0f;
                _rb.linearVelocity = Vector2.zero;
                _rb.angularVelocity = 0f;
                _rb.freezeRotation = true;
                _rb.mass = 0.0001f;
            }
            else
            {
                _isFlying = true;
                _rb.gravityScale = _movementStats.GravityScale;
                _rb.linearVelocity = Vector2.zero;
                _rb.angularVelocity = 0f;
                _rb.freezeRotation = false;
                _rb.mass = 1f;
            }
        }

        public void ApplyHitImpact(int hitDirection)
        {
            const float GRAVITY = -9.81f;

            SwitchControlType(false);

            // v = sqrt(2 * g * h) gives an apex of exactly _jumpHeight
            float gravity = Mathf.Abs(GRAVITY) * _movementStats.GravityScale;
            float launchSpeedY = Mathf.Sqrt(2f * gravity * _movementStats.JumpHeight);
            float launchSpeedX = _movementStats.HorizontalSpeed * hitDirection;

            Vector2 impulse = new Vector2(launchSpeedX, launchSpeedY);
            _rb.AddForce(impulse, ForceMode2D.Impulse);

            float spinDirection = UnityEngine.Random.value < 0.5f ? -1f : 1f;
            _rb.angularVelocity = _movementStats.Spin * spinDirection;
        }

        public void ResetVerticalVelocity()
        {
            _verticalVelocity = 0f;
        }

        #region Debug Methods
#if UNITY_EDITOR
        private void OnDrawGizmosSelected()
        {
            if (!_toggleGizmos) return;

            var velocity = Vector3.up * _verticalVelocity;

            GizmoUtils.DrawWireArrow(transform.position,
                                      _offset,
                                      velocity,
                                      _arrowWidth,
                                      _arrowHeight,
                                      _lineThickness,
                                      _verticalArrowColor);

            GizmoUtils.DrawWireArrow(transform.position,
                                      _offset,
                                      Vector3.right * _horizontalVelocity,
                                      _arrowWidth,
                                      _arrowHeight,
                                      _lineThickness,
                                      _horizontalArrowColor);

            if (_groundCheckPoint == null) return;

            Gizmos.color = Color.green;
            Gizmos.DrawWireCube(_groundCheckPoint.position, _movementStats.GroundCheckSize);

            Gizmos.color = Color.green;
            Gizmos.DrawWireCube(_ceilingCheckPoint.position, _movementStats.CeilingCheckSize);

            Gizmos.color = Color.green;
            Gizmos.DrawWireCube(_wallCheckPoint.position, _movementStats.WallCheckSize);
        }

        private void OnGUI()
        {
            if (!_showDebugLabels) return;

            GUIStyle style = new GUIStyle
            {
                fontSize = 30
            };
            style.normal.textColor = Color.white;

            var pivotX = 20f;
            var height = 50f;
            var width = 600f;

            GUI.Label(new Rect(pivotX, 20, width, height), "Gravity: " + _gravity, style);
            GUI.Label(new Rect(pivotX, 60, width, height), "Initial Jump Velocity: " + _initialJumpVelocity, style);
            GUI.Label(new Rect(pivotX, 100, width, height), "Vertical Velocity: " + _verticalVelocity, style);
            GUI.Label(new Rect(pivotX, 140, width, height), "Horizontal Velocity: " + _horizontalVelocity, style);
            GUI.Label(new Rect(pivotX, 180, width, height), "Rigidbody Linear Velocity: " + _rb.linearVelocity, style);
        }
#endif
        #endregion
    }
}

using System;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(PlayerController))]
    public class PlayerRenderer : MonoBehaviour
    {
        private static readonly int JumpID = Animator.StringToHash("isJumping");
        private static readonly int RunID = Animator.StringToHash("isRunning");
        private static readonly int FallID = Animator.StringToHash("isFalling");
        private static readonly int WallSlideID = Animator.StringToHash("isWallSliding");
        private static readonly int HitID = Animator.StringToHash("die");

        [SerializeField] private Animator _animator;
        [SerializeField] private SpriteRenderer _renderer;

        [SerializeField] private ParticleSystem _landFX;
        [SerializeField] private ParticleSystem _jumpFX;
        [SerializeField] private ParticleSystem _runFX;
        private int _defaultSortingOrder;

        private void Awake()
        {
            _defaultSortingOrder = _renderer.sortingOrder;
        }

        public void ResetHit()
        {
            _animator.ResetTrigger(HitID);
        }

        public void SlideWall(bool slide)
        {
            _animator.SetBool(WallSlideID, slide);
        }

        public void Land()
        {
            _animator.SetBool(FallID, false);
            _animator.SetBool(JumpID, false);
            PlayParticle(_landFX);
        }

        public void Fall()
        {
            _animator.SetBool(FallID, true);
            _animator.SetBool(JumpID, false);
        }

        public void Jump()
        {
            _animator.SetBool(JumpID, true);
            _animator.SetBool(FallID, false);
            PlayParticle(_jumpFX);
        }

        public void Turn(bool isRight)
        {
            _renderer.flipX = !isRight;
        }

        public void Move(bool isRunning)
        {
            _animator.SetBool(RunID, isRunning);

            if (isRunning)
            {
                if (!_runFX.isPlaying)
                    _runFX.Play();
            }
            else
            {

using System;
using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(PlayerInput), typeof(Rigidbody2D))]
    [RequireComponent(typeof(Collider2D))]
    public class PlayerController : MonoBehaviour, IHittable, IImpactable, IFollowTargetProvider
    {
        [Header("SFX Clips")]
        [SerializeField] private AudioClip _jumpClip;
        [SerializeField] private AudioClip _landClip;
        [SerializeField] private AudioClip _hurtClip;

        [SerializeField] private PlayerRenderer _renderer;
        [SerializeField] private PlayerMovement _movement;
        [SerializeField] private PlayerInput _input;

        [SerializeField] private float _bodyCenter = 0.786f; // got it from boxcollider
        [SerializeField] private Transform _cameraFollowTarget;
        [SerializeField] private AudioClip _playerHitSound;
        [SerializeField] private LayerMask _breakableMask;
        [SerializeField] private float _despawnMoveSpeed = 0.3f;

        private bool _canFollowTarget = true;
        private Collider2D _collider;
        private bool _isHittable = true;
        private bool _isImpactable = true;

        public Transform CameraFollowTarget { get { return _cameraFollowTarget; } }
        public int LookDirection { get { return _movement.LookDirection; } }
        public bool CanFollowTarget { get { return _canFollowTarget; } }
        public bool IsImpactable { get { return _isImpactable; } }

        public bool MovementEnabled { get; set; }

        public bool IsHittable { get { return _isHittable; } }

        public event Action OnFirstMove;

        #region Unity API Methods
        private void Awake()
        {
            _input = GetComponent<PlayerInput>();
            _collider = GetComponent<Collider2D>();
        }

        private void OnEnable()
        {
            _movement.OnMove += HandleMove;
            _movement.OnTurn += HandleTurn;
            _movement.OnFall += HandleFall;
            _movement.OnLand += HandleLand;
            _movement.OnJump += HandleJump;
            _movement.OnWallSlide += HandleWallSlide;

            _input.OnJumpPerformed += HandleJumpPerformed;
            _input.OnJumpCanceled += HandleJumpCanceled;
        }

        private void OnDisable()
        {
            _movement.OnMove -= HandleMove;
            _movement.OnTurn -= HandleTurn;
            _movement.OnFall -= HandleFall;
            _movement.OnLand -= HandleLand;
            _movement.OnJump -= HandleJump;
            _movement.OnWallSlide -= HandleWallSlide;

            _input.OnJumpPerformed -= HandleJumpPerformed;
            _input.OnJumpCanceled -= HandleJumpCanceled;
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            if ((_breakableMask & (1 << col.gameObject.layer)) == 0) return;

            if (!col.gameObject.TryGetComponent(out IBreakable breakable) || !breakable.IsBreakable)
                return;

            for (int i = 0; i < col.contactCount; i++)
            {
                float normalY = col.GetContact(i).normal.y;

                if (Mathf.Abs(normalY) <= 0.5f) continue; // ignore side contacts

                // normal points from the container towards the player,
                // so a negative Y means the player hit it from below
                if (normalY < 0f)
                    _movement.ResetVerticalVelocity();

                breakable.Break();
                return;
            }
        }

        private void Update()
        {
            _renderer.Move(_movement.IsRunning);

            if (!_movement.IsFlying) return;

            // below 0 on Y means the object has fallen off the bottom of the screen
            if (_movement.LinearVelocity.y < 0f && IsOutOfView())
            {
                _movement.SwitchControlType(true);
                GameEvents.Publish(new PlayerDiedEventData());

                gameObject.SetActive(false);
            }
        }

        private void FixedUpdate()
        {
            HandlePhysicsUpdate();
        }

        private void HandlePhysicsUpdate()
        {
            if (_movement.IsFlying) return;

            if (MovementEnabled)
            {
                _movement.SimulateMovement(_input.MoveInput);
            }

            _movement.ApplyMovement();
        }

        private bool IsOutOfView()
        {
            // viewport co-ordinate origin is (0,0) which is bottom left corner and top right corner is (1,1)
            // convert world position to viewport co-ordinate and then check if that value is less than 0 which is viewport bottom bound
            return Camera.main.WorldToViewportPoint(transform.position).y < 0f;
        }
        #endregion

        private void HandleMove()
        {
            OnFirstMove?.Invoke();
        }

        private void HandleTurn(bool turn)
        {
            _renderer.Turn(turn);
        }

        private void HandleFall()
        {
            _renderer.Fall();
        }

        private void HandleLand()
        {
            AudioManager.Instance.PlaySFX(_landClip);
            _renderer.Land();
        }

        private void HandleJump()
        {
            _renderer.Jump();
            AudioManager.Instance.PlaySFX(_jumpClip);
            OnFirstMove?.Invoke();
        }

        private void HandleWallSlide(bool slide)
        {
            _renderer.SlideWall(slide);
        }

        private void HandleJumpCanceled()
        {
            _movement.CutJump();
        }

        private void HandleJumpPerformed()
        {
            _movement.BufferJump();
        }

        #region Hit Methods
        public void ApplyImpact(GameObject source, float force)
        {
            _movement.PerformJump(force);

            if (source.TryGetComponent(out IHittable hittable) && hittable.IsHittable)
            {
                GameEvents.Publish(new StompedEnemyEventData());
                AudioManager.Instance.PlaySFX(_playerHitSound);
                hittable.TakeHit(transform.position);
            }
        }
        #endregion

        public void SetPosition(Vector3 pos)
        {
            pos.y -= _bodyCenter;
            transform.position = pos;
        }

        public void DespawnAt(Vector3 position, Action onComplete)
        {
            MovementEnabled = false;
            StartCoroutine(DespawnRoutine(position, onComplete));
            _renderer.Jump();
        }

        public void ResetState()
        {
            _movement.ResetState();
            _isImpactable = true;

            ResetHit();
            _isHittable = true;
            _renderer.ResetHit();

            _canFollowTarget = true;
        }

        public void ResetHit()
        {
            _movement.SwitchControlType(true);
            _collider.enabled = true;
            _renderer.SetDefaultOrder();
        }

        private IEnumerator DespawnRoutine(Vector3 position, Action onComplete)
        {
            position.y -= _bodyCenter;
            while (!_movement.StepTowards(position, _despawnMoveSpeed))
                yield return null;

            gameObject.SetActive(false);
            onComplete?.Invoke();
        }

        public void Spawn()
        {
            gameObject.SetActive(true);
            _renderer.Turn(true);
            _renderer.Fall();
            MovementEnabled = true;
            ResetState();
        }

        public void Despawn()
        {
            gameObject.SetActive(false);
        }

        public void ToggleInput(bool toggle)
        {
            _input.InputEnabled = toggle;
        }

        public void TakeHit(Vector2 hitSourcePosition)
        {
            _isHittable = false;
            MovementEnabled = false;
            GameEvents.Publish(new CameraShakeEventData());

            _collider.enabled = false;
            _renderer.SetSortingOrder(50);

            int hitDirection = transform.position.x - hitSourcePosition.x > 0f ? 1 : -1;

            _movement.ApplyHitImpact(hitDirection);

            _canFollowTarget = false;
            _isImpactable = false;
            _renderer.TriggerHit();

            AudioManager.Instance.PlaySFX(_hurtClip);
        }
    }
}

using UnityEngine;

namespace PixelPlatformer

    public interface IImpactable
    {
        bool IsImpactable { get; }
        void ApplyImpact(GameObject source, float force);
    }

    [RequireComponent(typeof(Collider2D))]
    public class StompImpactTrigger : MonoBehaviour
    {
        [SerializeField] private LayerMask _targetMask;
        [SerializeField] private float _normalThreshold = 0.5f;
        [SerializeField] private float _impactForce = 25f;

        private void OnCollisionEnter2D(Collision2D col)

            for (int i = 0; i < col.contactCount; i++)

                ContactPoint2D point = col.GetContact(i);

                // target above us
                if (-point.normal.y > _normalThreshold)
                {
                    if (col.gameObject.TryGetComponent(out IImpactable impactable) && impactable.IsImpactable)
                    {
                        impactable?.ApplyImpact(gameObject, _impactForce);
                    }
                    break;
                }
            }
        }
    }

using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Collider2D))]
    public class FallingPlatform : MonoBehaviour
    {
        [SerializeField] private LayerMask _offTriggerMask;
        [SerializeField] private SingleClipAnimator _onClipAnimator;
        [SerializeField] private SpriteRenderer _renderer;
        [SerializeField] private Sprite _offSprite;
        [SerializeField] private float _threshold = 0.5f;

        [SerializeField] private float _idleShakeAmount = 0.02f;
        [SerializeField] private float _warnShakeAmount = 0.06f;
        [SerializeField] private float _warnDuration = 0.4f;

        [SerializeField] private float _fallAcceleration = 1f;
        [SerializeField] private float _fallSpeed = 0.5f;

        private Collider2D _collider;
        private Vector3 _rendererStartLocalPos;
        private Coroutine _shakeRoutine;
        private float _currentFallSpeed;
        private bool _isOff;
        private WaitForSeconds _warnWait;

        private void Awake()
        {
            _collider = GetComponent<Collider2D>();
            _rendererStartLocalPos = _renderer.transform.localPosition;
            _warnWait = new WaitForSeconds(_warnDuration);
        }

        private void Start()
        {
            PlayOn();
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            if (_isOff)
                return;

            if ((_offTriggerMask.value & (1 << col.gameObject.layer)) == 0)
                return;

            Vector2 toTarget = -col.GetContact(0).normal;
            if (toTarget.y <= _threshold)
                return;

            PlayOff();
        }

        private void PlayOn()
        {
            _onClipAnimator.PlayClip();

            StartShake(_idleShakeAmount, -1f);
        }

        private void PlayOff()
        {
            _isOff = true;

            _onClipAnimator.StopClip();
            _renderer.sprite = _offSprite;

            StartCoroutine(OffRoutine());
        }

        private IEnumerator OffRoutine()
        {
            StartShake(_warnShakeAmount, _warnDuration);
            yield return _warnWait;

            _collider.enabled = false;

            yield return FallRoutine();
        }

        private void StartShake(float amount, float duration)
        {
            if (_shakeRoutine != null)
                StopCoroutine(_shakeRoutine);

            _shakeRoutine = StartCoroutine(ShakeRoutine(amount, duration));
        }

        private IEnumerator ShakeRoutine(float amount, float duration)
        {
            Transform t = _renderer.transform;
            float elapsed = 0f;

            while (duration < 0f || elapsed < duration)
            {
                elapsed += Time.deltaTime;

                Vector2 offset = Random.insideUnitCircle * amount;
                t.localPosition = _rendererStartLocalPos + (Vector3)offset;

                yield return null;
            }

            t.localPosition = _rendererStartLocalPos;
            _shakeRoutine = null;
        }

        private IEnumerator FallRoutine()
        {
            _currentFallSpeed = 0f;

            while (gameObject.activeSelf)
            {
                Vector2 pos = transform.position;

                _currentFallSpeed = Mathf.MoveTowards(_currentFallSpeed, _fallSpeed, _fallAcceleration * Time.deltaTime);
                pos.y -= _currentFallSpeed;

                transform.position = pos;

                if (Camera.main.WorldToViewportPoint(pos).y < 0)
                    gameObject.SetActive(false);

                yield return null;
            }
        }
    }
}

using UnityEngine;

namespace PixelPlatformer

    public class LoopingMovingPlatform : MovingPlatformBase
    {

        private bool _movingToEnd = true;
        private float _waitTimer;

        private void FixedUpdate()
        {
            if (_waitTimer > 0f)
            {
                _waitTimer -= Time.fixedDeltaTime;
                SetMoving(false);
                return;
            }

            Vector2 target = _movingToEnd ? _endPoint : _startPoint;

// ── FILE: MovingPlatformBase.cs ──
// Path: Assets/_Platformer/_Scripts/Traps/Platforms/MovingPlatformBase.cs

using UnityEngine;
#if UNITY_EDITOR
using UnityEditor;
using UnityEditor.SceneManagement;
#endif

namespace PixelPlatformer
{
    public enum PlatformAxis
    {
        Horizontal,
        Vertical
    }

    [RequireComponent(typeof(Rigidbody2D), typeof(Collider2D))]
    public abstract class MovingPlatformBase : MonoBehaviour
    {
        [Header("Path")]
        [SerializeField] protected PlatformAxis _axis = PlatformAxis.Horizontal;
        [SerializeField] protected float _distance = 4f;
        [SerializeField] protected float _speed = 2f;

        [Header("Animation")]
        [SerializeField] private SingleClipAnimator _moveClip;

        [Header("Chain Path")]
        [SerializeField] private Transform _pathRoot;
        [SerializeField] private GameObject _chainPrefab;
        [SerializeField] private float _chainSpacing = 0.5f;

        [Header("Rider")]
        [SerializeField] protected LayerMask _targetMask;
        [SerializeField, Range(0.1f, 1f)] private float _topContactThreshold = 0.5f;

        protected Rigidbody2D _rb;
        protected Vector2 _startPoint;
        protected Vector2 _endPoint;
        protected Vector2 _currentPosition;

        private Rigidbody2D _rider;
        private bool _isMoving;

        protected bool HasRider
        {
            get { return _rider != null; }
        }

        protected Vector2 EndOffset
        {
            get { return _axis == PlatformAxis.Horizontal ? new Vector2(_distance, 0f) : new Vector2(0f, _distance); }
        }

        protected virtual void Awake()
        {
            _rb = GetComponent<Rigidbody2D>();
            _rb.bodyType = RigidbodyType2D.Kinematic;
            _rb.interpolation = RigidbodyInterpolation2D.Interpolate;

            _startPoint = _rb.position;
            _endPoint = _startPoint + EndOffset;
            _currentPosition = _startPoint;

            _moveClip.monoBehaviour = this;
            if (_moveClip._renderer == null)
                _moveClip._renderer = GetComponentInChildren<SpriteRenderer>();

            if (_pathRoot != null && _pathRoot.IsChildOf(transform))
                _pathRoot.SetParent(transform.parent, true);
        }

        protected virtual void OnDisable()
        {
            _rider = null;
            _isMoving = false;
            _moveClip.StopClip();
        }

        protected void SetMoving(bool moving)
        {
            if (_isMoving == moving) return;

            _isMoving = moving;

            if (moving)
            {
                _moveClip.PlayClip();
                return;
            }

            _moveClip.StopClip();

            if (_moveClip._sprites != null && _moveClip._sprites.Length > 0)
                _moveClip._renderer.sprite = _moveClip._sprites[0];
        }

        protected bool StepTowards(Vector2 target, float speed)
        {
            Vector2 next = Vector2.MoveTowards(_currentPosition, target, speed * Time.fixedDeltaTime);
            Vector2 delta = next - _currentPosition;

            _currentPosition = next;
            _rb.position = next;

            SetMoving(delta != Vector2.zero);

            if (_rider != null)
                _rider.position += delta;

            return next == target;
        }

        private void OnCollisionEnter2D(Collision2D col)
        {
            EvaluateContact(col);
        }

        private void OnCollisionStay2D(Collision2D col)
        {
            EvaluateContact(col);
        }

        private void OnCollisionExit2D(Collision2D col)
        {
            if (col.rigidbody == _rider)
                _rider = null;
        }

        private void EvaluateContact(Collision2D col)
        {
            if ((_targetMask.value & (1 << col.gameObject.layer)) == 0) return;
            if (col.rigidbody == null) return;

            _rider = IsOnTop(col) ? col.rigidbody : null;
        }

        private bool IsOnTop(Collision2D col)
        {
            for (int i = 0; i < col.contactCount; i++)
            {
                if (-col.GetContact(i).normal.y > _topContactThreshold)
                    return true;
            }

            return false;
        }

        private void OnDrawGizmosSelected()
        {
            Vector2 start = Application.isPlaying ? _startPoint : (Vector2)transform.position;
            Vector2 end = Application.isPlaying ? _endPoint : start + EndOffset;

            Gizmos.color = Color.green;
            Gizmos.DrawLine(start, end);
            Gizmos.DrawWireSphere(start, 0.15f);
            Gizmos.DrawWireSphere(end, 0.15f);
        }

#if UNITY_EDITOR
        private void OnValidate()
        {
            _chainSpacing = Mathf.Max(0.1f, _chainSpacing);

            if (Application.isPlaying || EditorUtility.IsPersistent(this)) return;

            EditorApplication.delayCall -= RebuildChain;
            EditorApplication.delayCall += RebuildChain;
        }

        private void RebuildChain()
        {
            if (this == null || _pathRoot == null || _chainPrefab == null) return;

            for (int i = _pathRoot.childCount - 1; i >= 0; i--)
                DestroyImmediate(_pathRoot.GetChild(i).gameObject);

            float distance = EndOffset.magnitude;
            int segments = Mathf.Max(1, Mathf.RoundToInt(distance / _chainSpacing));
            int count = distance < 0.01f ? 1 : segments + 1;
            Vector3 start = transform.position;

            for (int i = 0; i < count; i++)
            {
                GameObject link = PrefabUtility.InstantiatePrefab(_chainPrefab, _pathRoot) as GameObject;
                if (link == null) break;

                float t = segments > 0 ? (float)i / segments : 0f;
                link.transform.position = start + (Vector3)EndOffset * t;
            }

            EditorSceneManager.MarkSceneDirty(gameObject.scene);
        }
#endif
    }
}

using UnityEngine;

namespace PixelPlatformer
{
    public class Spikes : MonoBehaviour
    {
        [SerializeField] private LayerMask _targetMask;

        private void OnCollisionEnter2D(Collision2D col)
        {
            if ((_targetMask.value & (1 << col.gameObject.layer)) == 0)
                return;

            if (!col.collider.TryGetComponent<IHittable>(out var hittable)
                || !hittable.IsHittable)
                return;

            hittable.TakeHit(transform.position);
        }
    }
}

using System;
using System.Collections.Generic;
using DG.Tweening;
using UnityEngine;
using UnityEngine.UI;

namespace PixelPlatformer
{
    [RequireComponent(typeof(RectTransform))]
    public class LevelSelectScreen : MonoBehaviour
    {
        [SerializeField] private CanvasGroup _canvasGroup;

        [SerializeField] private Button _backButton;
        [SerializeField] private LevelSlot _slotPrefab;
        [SerializeField] private RectTransform _slotContainer;
        [SerializeField] private float _fadeDuration = 0.5f;
        private readonly List<LevelSlot> _slots = new List<LevelSlot>();

        private Tween _fadeTween;

        public event Action BackRequested;

        private void Start()
        {
            Hide();
        }

        private void OnEnable()
        {
            _backButton.onClick.AddListener(HandleBackClicked);
        }

        private void OnDisable()
        {
            _backButton.onClick.RemoveListener(HandleBackClicked);
        }

        private void OnDestroy()
        {
            for (int i = 0; i < _slots.Count; i++)
            {
                _slots[i].Clicked -= HandleSlotClicked;
            }
        }

        public void Init(List<LevelRecord> levelRecords)
        {
            for (int i = 0; i < levelRecords.Count; i++)
            {
                var slot = Instantiate(_slotPrefab, _slotContainer);

                slot.Clicked += HandleSlotClicked;

                Sprite levelIcon = GameManager.Instance.GetLevelIcon(levelRecords[i].levelIndex);

                slot.BindData(levelRecords[i].levelIndex,
                              levelIcon,
                               levelRecords[i].isUnlocked,
                               levelRecords[i].bestScore,
                              levelRecords[i].StarsEarned);

                _slots.Add(slot);
            }
        }

        public void Open(Action onComplete = null)
        {
            _fadeTween?.Kill();
            _canvasGroup.interactable = false;
            _canvasGroup.blocksRaycasts = true;

            _fadeTween = _canvasGroup.DOFade(1f, _fadeDuration).SetLink(gameObject).OnComplete(() =>
            {
                _canvasGroup.interactable = true;
                onComplete?.Invoke();
            });
        }

        public void Close(Action onComplete = null)
        {
            _fadeTween?.Kill();
            _canvasGroup.interactable = false;

            _fadeTween = _canvasGroup.DOFade(0f, _fadeDuration).SetLink(gameObject).OnComplete(() =>
            {
                _canvasGroup.blocksRaycasts = false;
                onComplete?.Invoke();
            });
        }

        public void Refresh(List<LevelRecord> levelRecords)
        {
            for (int i = 0; i < _slots.Count && i < levelRecords.Count; i++)
            {
                Sprite levelIcon = GameManager.Instance.GetLevelIcon(levelRecords[i].levelIndex);

                _slots[i].BindData(levelRecords[i].levelIndex,
                                   levelIcon,
                                   levelRecords[i].isUnlocked,
                                   levelRecords[i].bestScore,
                                   levelRecords[i].StarsEarned);
            }
        }

        private void Hide()
        {
            _canvasGroup.alpha = 0f;
            _canvasGroup.interactable = false;
            _canvasGroup.blocksRaycasts = false;
        }

        private void HandleBackClicked()
        {
            BackRequested?.Invoke();
        }

        private void HandleSlotClicked(int levelIndex)
        {
            GameManager.Instance.StartLevel(levelIndex);
        }
    }
}


// ── FILE: LevelSlot.cs ──

using System;
using TMPro;
using UnityEngine;
using UnityEngine.UI;

namespace PixelPlatformer
{
    public class LevelSlot : MonoBehaviour
    {
        [SerializeField] private Button _button;
        [SerializeField] private Image _levelIcon;
        [SerializeField] private TMP_Text _bestScore;

        [SerializeField] private Image[] _starImages;
        [SerializeField] private Sprite _lockSprite;
        [SerializeField] private Sprite _starOnSprite;
        [SerializeField] private Sprite _starOffSprite;
        private int _index;

        public bool IsSelectable
        {
            get
            {
                return _button.interactable;
            }
        }

        public event Action<int> Clicked;

        private void OnEnable()
        {
            _button.onClick.AddListener(HandleClick);
        }

        private void OnDisable()
        {
            _button.onClick.RemoveListener(HandleClick);
        }

        public void BindData(int index, Sprite levelSprite, bool unlocked, float bestScore, int stars)
        {
            _index = index;

            _levelIcon.sprite = unlocked ? levelSprite : _lockSprite;
            _bestScore.gameObject.SetActive(unlocked);
            _levelIcon.enabled = _levelIcon.sprite != null;

            _button.interactable = unlocked;

            if (unlocked)
            {
                for (int i = 0; i < _starImages.Length; i++)
                    _starImages[i].sprite = i < stars ? _starOnSprite : _starOffSprite;

                _bestScore.text = "Best Score: " + bestScore.ToString();
            }
            else
            {
                for (int i = 0; i < _starImages.Length; i++)
                    _starImages[i].sprite = _starOffSprite;
            }
        }

        public void Select()
        {
            if (_button.interactable)
                _button.Select();
        }

        private void HandleClick()
        {
            Clicked?.Invoke(_index);
        }
    }
}

        [SerializeField] private Slider _sFXVolumeSlider;
        [SerializeField] private Slider _musicVolumeSlider;
        [SerializeField] private Button _eraseButton;
        [SerializeField] private PanelScaleTween _panelScaleTween;
        [SerializeField] private Button _closeButton;

        public event Action EraseRequested;
        public event Action OnCloseClicked;

        private void OnEnable()
        {
            _closeButton.onClick.AddListener(HandleCloseClicked);
        }

        private void OnDisable()
        {
            _closeButton.onClick.RemoveListener(HandleCloseClicked);
        }

        public void PlayOpen(Action onComplete = null)
        {
            _sFXVolumeSlider.onValueChanged.AddListener(SetSFXVolume);
            _musicVolumeSlider.onValueChanged.AddListener(SetMusicVolume);
            _eraseButton.onClick.AddListener(EraseProgress);

                onComplete?.Invoke();

            _sFXVolumeSlider.onValueChanged.RemoveListener(SetSFXVolume);
            _musicVolumeSlider.onValueChanged.RemoveListener(SetMusicVolume);

            AudioManager.Instance.SaveData();

            _panelScaleTween.PlayClose(() =>

                onComplete?.Invoke();
            });

        {
            PlayClose();
            OnCloseClicked?.Invoke();
        }

        {
            AudioManager.Instance.SetSFXVolume(value);
        }

        private void SetMusicVolume(float value)
        {
            AudioManager.Instance.SetMusicVolume(value);
        }

        private void EraseProgress()
        {
            EraseRequested?.Invoke();
        }

        private void SetSliders()
        {
            _sFXVolumeSlider.SetValueWithoutNotify(AudioManager.Instance.SFXVolume);
            _musicVolumeSlider.SetValueWithoutNotify(AudioManager.Instance.MusicVolume);
        }
    }

using System;
using UnityEngine;
using UnityEngine.UI;

public abstract class TabGroup<T, K> : MonoBehaviour where T : Button where K : MonoBehaviour 
{
    [SerializeField] protected Tab<T, K>[] _tabs;

    public event Action<K> OnSelectTab;

    protected virtual void OnEnable()
    {
        foreach (var item in _tabs)
        {
            item.tabButton.onClick.AddListener(() => HandleSelectTab(item.tabPanel));
        }
    }

    protected virtual void OnDisable()
    {
        foreach (var item in _tabs)
        {
            item.tabButton.onClick.RemoveListener(() => HandleSelectTab(item.tabPanel));
        }
    }

    protected virtual void HandleSelectTab(K tabPanel)
    {
        OnSelectTab?.Invoke(tabPanel);
    }
}

using System;
using TMPro;
using UnityEngine;
using UnityEngine.UI;
using UnityEngine.Video;

namespace PixelPlatformer
{
    public class TutorialPanel : MonoBehaviour
    {
        [Serializable]
        private class ClipLabelPair
        {
            public VideoClip clip;
            public string label;
        }

        [SerializeField] private ClipLabelPair[] _clipLabelPairs;
        [SerializeField] private VideoPlayer _videoPlayer;

        [SerializeField] private PanelScaleTween _panelScaleTween;
        [SerializeField] private Button _previousButton;
        [SerializeField] private Button _nextButton;
        [SerializeField] private Button _closeButton;
        [SerializeField] private TMP_Text _guideLabel;

        private int _currentClipIndex;
        private bool _isOpen;

        public event Action OnOpened;
        public event Action OnClosed;

        public void Init()
        {
            _isOpen = false;
            SetClip(0);
            _videoPlayer.Stop();
        }

        private void OnEnable()
        {
            _previousButton.onClick.AddListener(PlayPreviousClip);
            _nextButton.onClick.AddListener(PlayNextClip);
            _closeButton.onClick.AddListener(Hide);
        }

        private void OnDisable()
        {
            _previousButton.onClick.RemoveListener(PlayPreviousClip);
            _nextButton.onClick.RemoveListener(PlayNextClip);
            _closeButton.onClick.RemoveListener(Hide);
        }

        public void Show()
        {
            if (_isOpen) return;
            _isOpen = true;

            OnOpened?.Invoke();

            _videoPlayer.Play();
            _panelScaleTween.PlayOpen();
        }

        public void Hide()
        {
            if (!_isOpen) return;
            _isOpen = false;

            OnClosed?.Invoke();

            _panelScaleTween.PlayClose(() =>
            {
                _videoPlayer.Stop();
            });
        }

        public void Toggle()
        {
            if (_isOpen)
                Hide();
            else
                Show();
        }

        private void SetClip(int index)
        {
            _currentClipIndex = index;
            _videoPlayer.clip = _clipLabelPairs[index].clip;

            _guideLabel.text = _clipLabelPairs[index].label;

            _previousButton.gameObject.SetActive(index > 0);
            _nextButton.gameObject.SetActive(index < _clipLabelPairs.Length - 1);
        }

        private void PlayClip(int index)
        {
            SetClip(index);
            _videoPlayer.Play();
        }

        private void PlayNextClip()
        {
            PlayClip(_currentClipIndex + 1);
        }

        private void PlayPreviousClip()
        {
            PlayClip(_currentClipIndex - 1);
        }
    }
}

using DG.Tweening;
using UnityEngine;
using UnityEngine.Video;

namespace PixelPlatformer
{
    public class TutorialManager : MonoBehaviour
    {
        [SerializeField] private VideoClip[] _tutorialClips;
        [SerializeField] private GameObject _guideRoot;
        [SerializeField] private Ease _showTweenEase = Ease.OutCubic;
        [SerializeField] private float _showTweenDuration = 0.5f;

        private RectTransform _guideRootRect;
        private Vector2 _originalAnchoredPosition;

        private void Awake()
        {
            _guideRootRect = _guideRoot.GetComponent<RectTransform>();
            _originalAnchoredPosition = _guideRootRect.anchoredPosition;
            _guideRoot.SetActive(false);
        }

        public void ShowGuide(Vector3 anchoredPos)
        {
            _guideRootRect.anchoredPosition = anchoredPos;
            _guideRootRect.localScale = Vector3.zero;
            _guideRoot.SetActive(true);

            _guideRootRect.DOAnchorPos(_originalAnchoredPosition, _showTweenDuration).SetEase(_showTweenEase);
            _guideRootRect.DOScale(Vector3.one, _showTweenDuration).SetEase(_showTweenEase);
        }

        public void HideGuide(Vector3 anchoredPos)
        {
            _guideRootRect.DOAnchorPos(anchoredPos, _showTweenDuration).SetEase(_showTweenEase);
            _guideRootRect.DOScale(Vector3.zero, _showTweenDuration).SetEase(_showTweenEase)
                .OnComplete(() => _guideRoot.SetActive(false));
        }

        public void ToggleGuide(Vector3 anchoredPos)
        {
            if (_guideRoot.activeSelf)
                HideGuide(anchoredPos);
            else
                ShowGuide(anchoredPos);
        }
    }
}

using DG.Tweening;
using UnityEngine;
using UnityEngine.EventSystems;
using UnityEngine.UI;

namespace PixelPlatformer
{
    [RequireComponent(typeof(Button), typeof(RectTransform))]
    public class ButtonFeedback : MonoBehaviour, IPointerDownHandler, IPointerUpHandler, IPointerExitHandler, ISubmitHandler
    {
        [SerializeField] private ButtonFeedbackConfigSO _config;

        private RectTransform _rect;
        private Button _button;
        private Vector3 _originalScale;
        private Tween _tween;
        private bool _isPressed;

        private void Awake()
        {
            _button = GetComponent<Button>();
            _rect = GetComponent<RectTransform>();

            if (_rect == null)
                _rect = (RectTransform)transform;

            _originalScale = _rect.localScale;
        }

        private void OnDisable()
        {
            _tween?.Kill();
            _rect.localScale = _originalScale;
            _isPressed = false;
        }

        public void OnPointerDown(PointerEventData eventData)
        {
            if (!_button.IsInteractable()) return;

            _isPressed = true;
            PlayClick();
            ScaleTo(_originalScale * _config.PressedScale, _config.PressDuration, _config.PressEase);
        }

        public void OnPointerUp(PointerEventData eventData)
        {
            Release();
        }

        public void OnPointerExit(PointerEventData eventData)
        {
            Release();
        }

        public void OnSubmit(BaseEventData eventData)
        {
            if (!_button.IsInteractable()) return;
            PlayClick();
        }

        private void Release()
        {
            if (!_isPressed) return;

            _isPressed = false;
            ScaleTo(_originalScale, _config.ReleaseDuration, _config.ReleaseEase);
        }

        private void ScaleTo(Vector3 scale, float duration, Ease ease)
        {
            _tween?.Kill();
            _tween = _rect.DOScale(scale, duration)
                .SetEase(ease)
                .SetUpdate(true)
                .SetLink(gameObject);
        }

        private void PlayClick()
        {
            if (_config.ClickClip != null)
                AudioManager.Instance.PlaySFX(_config.ClickClip);
        }
    }
}

using TMPro;
using UnityEngine;

namespace PixelPlatformer
{
    public class HUDController : MonoBehaviour
    {
        [SerializeField] private TMP_Text _enemiesDefeatedText;
        [SerializeField] private TMP_Text _fruitsCollectedCountText;
        [SerializeField] private TMP_Text _timerText;

        private float _elapsedTime;
        private bool _isTimeTicking;

        public float ElapsedTime { get { return _elapsedTime; } }

        public void SetCollectibles(int count)
        {
            _fruitsCollectedCountText.text = count.ToString();
        }

        public void SetEnemiesDefeated(int count)
        {
            _enemiesDefeatedText.text = count.ToString();
        }

        private void Update()
        {
            if (!_isTimeTicking) return;

            _elapsedTime += Time.unscaledDeltaTime; // use unscaled timeScale, so pausing doesn't freeze it
            SetTime(_elapsedTime);
        }

        public void SetTime(float seconds)
        {
            int mins = (int)(seconds / 60f);
            float secs = seconds - mins * 60f;
            _timerText.text = $"{mins:00}:{secs:00.00}";
        }

        public void StopTimer()
        {
            _isTimeTicking = false;
        }

        public void StartTimer()
        {
            _isTimeTicking = true;
            _elapsedTime = 0f;
        }
    }
}

using System;
using DG.Tweening;
using UnityEngine;

namespace PixelPlatformer
{
    public class PanelScaleTween : MonoBehaviour
    {
        [SerializeField] private CanvasGroup _contentCanvasgroup;
        [SerializeField] private PanelScaleTweenConfigSO _config;
        [SerializeField] private RectTransform _rect;

        private void Start()
        {
            _contentCanvasgroup.alpha = 0f;
            _contentCanvasgroup.interactable = false;
            _contentCanvasgroup.blocksRaycasts = false;
        }

        public void PlayOpen(Action onComplete = null)
        {
            _contentCanvasgroup.alpha = 1f;
            _contentCanvasgroup.interactable = false;
            _contentCanvasgroup.blocksRaycasts = true;

            _rect.localScale = Vector3.one * _config.StartScale;

            _rect.DOKill();
            _rect.DOScale(1f, _config.OpenDuration)
                 .SetEase(_config.OpenEase)
                 .SetLink(_rect.gameObject)
                 .OnComplete(() =>
                 {
                     _contentCanvasgroup.interactable = true;
                     onComplete?.Invoke();
                 });
        }

        public void PlayClose(Action onComplete = null)
        {
            _contentCanvasgroup.interactable = false;

            _rect.DOKill();
            _rect.DOScale(_config.StartScale, _config.CloseDuration)
                 .SetEase(_config.CloseEase)
                 .SetLink(_rect.gameObject)
                 .OnComplete(() =>
                 {
                     _contentCanvasgroup.blocksRaycasts = false;
                     _contentCanvasgroup.alpha = 0f;

                     onComplete?.Invoke();
                 });
        }

#if UNITY_EDITOR
        [ContextMenu("Test Open")]
        private void TestOpen()
        {
            PlayOpen(() =>
            {
                Debug.Log("Open complete", this);
            });
        }

        [ContextMenu("Test Close")]
        private void TestClose()
        {

            {
                Debug.Log("Close complete", this);
            });
        }

    }

    {

        [SerializeField] private CanvasGroup _group;
        [SerializeField] private TMP_Text _label;

        [SerializeField] private float _holdTime = 2f;
        [SerializeField] private float _fadeTime = 0.25f;
        [SerializeField] private float _slideDistance = 60f;

        private Vector2 _shownPos;
        private Sequence _seq;

        private void Awake()
        {
            _shownPos = _rect.anchoredPosition;
            _group.alpha = 0f;
        }

        private void OnDestroy()
        {
            _seq?.Kill();
        }

        public void Show(string message)
        {
            _seq?.Kill();

            _label.text = message;
            _group.alpha = 0f;
            _rect.anchoredPosition = _shownPos - new Vector2(0f, _slideDistance);

            _seq = DOTween.Sequence()
                          .Append(_group.DOFade(1f, _fadeTime))
                          .Join(_rect.DOAnchorPos(_shownPos, _fadeTime).SetEase(Ease.OutBack))
                          .AppendInterval(_holdTime)
                          .Append(_group.DOFade(0f, _fadeTime))
                          .SetLink(gameObject);
        }
    }

        [SerializeField] private TMP_Text _totalScoreText;
        [SerializeField] private Color _bestTextColor = Color.blue;

        [Header("Buttons")]
        [SerializeField] private Button _replayButton;
        [SerializeField] private Button _goToNextLevelButton;
        [SerializeField] private Button _mainMenuButton;

        [SerializeField] private Sprite[] _sprites;
        [SerializeField] private Sprite _emptyStarSprite;
        [SerializeField] private int _frameRate = 12;
        [SerializeField] private PanelScaleTween _panelScaleTween;

        private int _keyFrame;
        private Coroutine _coroutine;
        private int _totalEnemies;
        private int _totalFruits;

        {
            _replayButton.onClick.AddListener(PlayAgain);
            _goToNextLevelButton.onClick.AddListener(NextLevel);
            _mainMenuButton.onClick.AddListener(MainMenu);
        }

        private void OnDisable()
        {
            _replayButton.onClick.RemoveListener(PlayAgain);
            _goToNextLevelButton.onClick.RemoveListener(NextLevel);
            _mainMenuButton.onClick.RemoveListener(MainMenu);
        }

        public void Setup(Sprite levelIcon, int totalEnemies, int totalFruits)
        {
            _levelIcon.sprite = levelIcon;
            _totalEnemies = totalEnemies;
            _totalFruits = totalFruits;
        }

        public void Open(LevelResult result, Action onComplete = null)
        {
            _timeTakenText.text = $"{result.timeTaken:0.0}s";
            _enemiesDefeated.text = $"{result.enemiesDefeated}/{_totalEnemies}";
            _fruitsCollectedText.text = $"{result.fruitsCollected}/{_totalFruits}";
            _totalScoreText.text = result.isNewBest ? $"<color=#{ColorUtility.ToHtmlStringRGB(_bestTextColor)}>Best! </color>{result.bestScore}" : result.score.ToString();

            foreach (Image image in _starImages)
            {
                image.sprite = _emptyStarSprite;
            }

            AudioManager.Instance.PlaySFX(_levelCompleteClip);

            _panelScaleTween.PlayOpen(() =>
            {
                PlayStarPopupClip(result.starsEarned);
                onComplete?.Invoke();
            });
        }

        public void Close(Action onComplete = null)
        {
            _panelScaleTween.PlayClose(() =>
            {
                onComplete?.Invoke();
            });
        }

        public void PlayStarPopupClip(int stars, Action onComplete = null)
        {
            if (_coroutine != null)
                StopClip();

            _coroutine = StartCoroutine(PlayClipRoutine(stars, onComplete));
        }

        public void StopClip()
        {
            if (_coroutine == null) return;

            StopCoroutine(_coroutine);
            _coroutine = null;
        }

        private IEnumerator PlayClipRoutine(int stars, Action onComplete = null)
        {
            for (int i = 0; i < stars; i++)
            {
                _keyFrame = 0;
                var elapsedTime = 0f;
                var targetDeltaTime = 1f / _frameRate;

                // for every 1/12the time i.e 12fps considering default framerate, jump to next keyframe
                while (_keyFrame < _sprites.Length)
                {
                    elapsedTime += Time.deltaTime;

                    if (elapsedTime >= targetDeltaTime)
                    {
                        _starImages[i].sprite = _sprites[_keyFrame];
                        _keyFrame++;
                        elapsedTime -= targetDeltaTime;
                    }

                    yield return null;
                }
            }
        }

        {
            GameManager.Instance.RestartLevel();
        }

        {
            Close(() => LevelManager.Instance.GoToNextLevel());
        }

        private void MainMenu()
        {
            GameManager.Instance.GoToMainMenu();
        }
    }

using UnityEngine;
using UnityEngine.Tilemaps;

namespace PixelPlatformer
{
    [RequireComponent(typeof(TilemapRenderer), typeof(Tilemap))]
    public class AnimatedBackground : MonoBehaviour
    {
        [SerializeField] private float _scrollSpeed = 2.5f;
        [SerializeField] private TilemapRenderer _renderer;
        [SerializeField] private Tilemap _map;
        [SerializeField] private TileBase _targetTile;
        [SerializeField] private bool _compressBounds = true;

        private float _scrollDownPos;
        private Transform _camTransform;

        public void Setup(Transform camTransform)
        {
            if (_compressBounds)
                _map.CompressBounds();

            SwapTiles(_targetTile);

            _camTransform = camTransform;
        }

        private void Update()
        {
            Move();
        }

        private void Move()
        {
            if (!_camTransform) return;

            float camHorizontalPos = _camTransform.position.x;
            Vector2 pos = transform.position;

            _scrollDownPos -= _scrollSpeed * Time.deltaTime;
            _scrollDownPos = Mathf.Repeat(_scrollDownPos, _renderer.bounds.extents.y / 2f);

            int targetHorizontalPos = (int)camHorizontalPos / ((int)transform.localScale.x * (int)_map.layoutGrid.cellSize.x);

            pos.x = targetHorizontalPos * transform.localScale.x * _map.layoutGrid.cellSize.x;
            pos.y = _scrollDownPos;

            transform.position = pos;
        }

        public void SwapTiles(TileBase newtile)
        {
            foreach (Vector3Int position in _map.cellBounds.allPositionsWithin)
            {
                if (_map.HasTile(position))
                {
                    _map.SetTile(position, newtile);
                }
            }
        }
    }
}

// ── FILE: GameplayManager.cs ──

using System;
using UnityEngine;

namespace PixelPlatformer
{
    public enum GameplayState
    {
        None,
        Start,
        Play,
        Pause,
        Lose,
        Win
    }


    public class GameplayManager : Singleton<GameplayManager>
    {
        [SerializeField] private TutorialPanel _tutorialPanel;
        [SerializeField] private PlayerManager _playerManager;
        [SerializeField] private HUDController _hUDController;
        private LevelManager _levelManager;

        private bool _isTutorialOpen;
        private bool _tutorialWasInputEnabled;

        public float ElapsedTime { get { return _hUDController.ElapsedTime; } }

        public GameplayState State { get; private set; } = GameplayState.None;

        private void OnEnable()
        {
            _tutorialPanel.OnOpened += HandleTutorialOpened;
            _tutorialPanel.OnClosed += HandleTutorialClosed;

            GameEvents.Subscribe<PlayerDiedEventData>(HandlePlayerDied);
            GameEvents.Subscribe<LevelCompletedEvent>(HandleLevelCompleted);
        }

        private void OnDisable()
        {
            _tutorialPanel.OnOpened -= HandleTutorialOpened;
            _tutorialPanel.OnClosed -= HandleTutorialClosed;

            GameEvents.Unsubscribe<PlayerDiedEventData>(HandlePlayerDied);
            GameEvents.Unsubscribe<LevelCompletedEvent>(HandleLevelCompleted);
        }

        public void Initialize(int levelIndex)
        {
            State = GameplayState.Start;

            _isTutorialOpen = false;
            _tutorialWasInputEnabled = false;

            _tutorialPanel.Init();

            _playerManager = PlayerManager.Instance;
            _playerManager.ToggleInput(false);

            _levelManager = LevelManager.Instance;
            _levelManager.SetupLevel(ProgressionManager.Instance.GetRecord(levelIndex));
        }

        private void HandleTutorialOpened()
        {
            _isTutorialOpen = true;

            _tutorialWasInputEnabled = _playerManager.InputEnabled;

            _playerManager.ToggleInput(false);
        }

        private void HandleTutorialClosed()
        {
            _isTutorialOpen = false;

            if (State == GameplayState.Play && _tutorialWasInputEnabled)
                _playerManager.ToggleInput(true);

            _tutorialWasInputEnabled = false;
        }

        public void StartSession()
        {
            if (_levelManager == null)
            {
                Debug.LogError("Cannot start session. LevelManager is missing.");
                return;
            }

            State = GameplayState.Play;

            _levelManager.StartLevel();
        }

        public void EndSession()
        {
            StopTimer();

            if (_playerManager != null)
                _playerManager.ToggleInput(false);

            State = GameplayState.None;
        }

        public void ResumeGame()
        {
            if (State != GameplayState.Pause)
                return;

            State = GameplayState.Play;

            if (!_isTutorialOpen)
                _playerManager.ToggleInput(true);

            UIManager.Instance.HidePauseMenu();
        }

        public void PauseGame()
        {
            if (State != GameplayState.Play)
                return;

            State = GameplayState.Pause;

            _playerManager.ToggleInput(false);
            UIManager.Instance.ShowPauseMenu();
        }


        public void StartTimer()
        {
            _hUDController.StartTimer();
        }

        public void StopTimer()
        {
            _hUDController.StopTimer();
        }

        private void HandleLevelCompleted(LevelCompletedEvent data)
        {
            if (State != GameplayState.Play)
                return;

            State = GameplayState.Win;

            StopTimer();
            _playerManager.ToggleInput(false);

            UIManager.Instance.ShowWinScreen(data.levelResult);
        }

        private void HandlePlayerDied(PlayerDiedEventData data)
        {
            if (State != GameplayState.Play)
                return;

            State = GameplayState.Lose;

            StopTimer();
            _playerManager.ToggleInput(false);

            UIManager.Instance.ShowLoseScreen();
        }
    }
}

using System;
using Cysharp.Threading.Tasks;
using DG.Tweening;
using UnityEngine;

namespace PixelPlatformer
{
    [Serializable]
    public struct TransitionSpritesColumn
    {
        [SerializeField] private RectTransform[] _spriteRects;

        public void ResetState()
        {
            foreach (RectTransform rect in _spriteRects)
                rect.localScale = Vector3.zero;
        }

        public Sequence PopUp(float duration)
        {
            Sequence sequence = DOTween.Sequence();

            foreach (RectTransform rect in _spriteRects)
                sequence.Join(rect.DOScale(Vector3.one, duration));

            return sequence;
        }

        public Sequence PopDown(float duration)
        {
            Sequence sequence = DOTween.Sequence();

            foreach (RectTransform rect in _spriteRects)
                sequence.Join(rect.DOScale(Vector3.zero, duration));

            return sequence;
        }
    }

    public class SceneTransitionFX : MonoBehaviour
    {
        [Header("Transition")]
        [SerializeField] private TransitionSpritesColumn[] _columns;

        [SerializeField] private float _columnDelay = 0.1f;
        [SerializeField] private float _tweenDuration = 0.5f;

        [Header("Canvas")]
        [SerializeField] private CanvasGroup _canvasGroup;

        private Sequence _transitionSequence;

        public void ResetState()
        {
            foreach (TransitionSpritesColumn column in _columns)
                column.ResetState();
        }

        public void SetVisibility(bool isVisible)
        {
            _canvasGroup.alpha = isVisible ? 1f : 0f;
            _canvasGroup.interactable = isVisible;
            _canvasGroup.blocksRaycasts = isVisible;
        }

        [ContextMenu("Play In")]
        public async UniTask PlayIn()
        {
            KillTransition();

            SetVisibility(true);

            _transitionSequence = DOTween.Sequence();

            for (int i = 0; i < _columns.Length; i++)
            {
                _transitionSequence.Insert(
                    i * _columnDelay,
                    _columns[i].PopUp(_tweenDuration)
                );
            }

            await _transitionSequence.AsyncWaitForCompletion();
        }

        [ContextMenu("Play Out")]
        public async UniTask PlayOut(Action onComplete = null)
        {
            KillTransition();

            _transitionSequence = DOTween.Sequence();

            for (int i = 0; i < _columns.Length; i++)
            {
                _transitionSequence.Insert(
                    i * _columnDelay,
                    _columns[i].PopDown(_tweenDuration)
                );
            }

            _transitionSequence.OnComplete(() =>
            {
                onComplete?.Invoke();
            });

            await _transitionSequence.AsyncWaitForCompletion();

            SetVisibility(false);
        }

        private void KillTransition()
        {
            _transitionSequence?.Kill();
            _transitionSequence = null;
        }

        private void OnDestroy()
        {
            KillTransition();
        }
    }
}

// ── FILE: SingleClipAnimator.cs ──
// Path: Assets/_Platformer/_Scripts/SingleClipAnimator.cs

using System;
using System.Collections;
using UnityEngine;

namespace PixelPlatformer
{
    [Serializable]
    public class SingleClipAnimator
    {
        public MonoBehaviour monoBehaviour;
        public SpriteRenderer _renderer;
        public Sprite[] _sprites;
        public bool _loop;
        public int _frameRate = 12;
        private int _keyFrame;
        private Coroutine _coroutine;

        public event Action OnComplete;

        public void PlayClip(Action onComplete = null)
        {
            if (_coroutine != null)
                StopClip();

            _coroutine = monoBehaviour.StartCoroutine(PlayClipRoutine(onComplete));
        }

        public void StopClip()
        {
            if(_coroutine == null) return;
            
            monoBehaviour.StopCoroutine(_coroutine);
            _coroutine = null;
        }

        private IEnumerator PlayClipRoutine(Action onCompleteAction = null)
        {
            _keyFrame = 0;
            var elapsedTime = 0f;
            var targetDeltaTime = 1f / _frameRate;

            // for every 1/12the time i.e 12fps considering default framerate, jump to next keyframe
            while (_keyFrame < _sprites.Length)
            {
                elapsedTime += Time.deltaTime;

                if (elapsedTime >= targetDeltaTime)
                {
                    _renderer.sprite = _sprites[_keyFrame];
                    _keyFrame++;
                    elapsedTime -= targetDeltaTime;
                }

                if (_loop)
                    _keyFrame %= _sprites.Length;

                yield return null;
            }

            onCompleteAction?.Invoke();
            OnComplete?.Invoke();
        }
    }
}
